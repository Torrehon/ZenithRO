ein_in01,18,28,4	script	Guildsman#BLS	731,{
	mes "[Guildman Altisen]";
	if (Upper == 1) {
		mes "Clang! Clang! Ah, a Transcendent craftsman stands before me!";
		next;
		mes "[Guildman Altisen]";
		mes "May your hammer strike true and your forged steel never shatter in battle.";
		close;
	}
	if (BaseJob != Job_Merchant) {
		if (BaseJob == Job_Blacksmith) {
			mes "Greetings, master smith! The furnaces of Einbroch burn brightly in honor of your craft!";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to the Blacksmith Guild! If you wish to forge weapons, train first as a Merchant in Alberta.";
			close;
		}
		else {
			mes "Welcome to the forge hall! Mind the flying sparks and hot iron.";
			close;
		}
	}

	mes "Welcome to the legendary Blacksmith Guild of Einbroch!";
	mes "Here, amidst blazing fires and heavy anvils, our guild masters turn raw iron and rare ores into deadly weaponry and indestructible armor.";
	next;
	mes "[Guildman Altisen]";
	mes "To earn the rank of ^0000FFBlacksmith^000000, a Merchant must demonstrate deep passion for metallurgy and battle forging.";
	next;
	mes "[Guildman Altisen]";
	mes "Our guild requires two credentials:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Merchant.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), confirming your prestige across Rune-Midgard.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Guildman Altisen]";
		mes "Let me check your hammer and credentials...";
		next;
		if (JobLevel < 50) {
			mes "[Guildman Altisen]";
			mes "Your Job Level is currently " + JobLevel + ". Keep trading and training until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Guildman Altisen]";
			mes "You do not have a ^FF0000Proof of Valor^000000. Bring this token of merit from the realm's associations to complete your induction.";
		}
		close;
	}

	mes "[Guildman Altisen]";
	mes "Outstanding! You have attained Job Level 50 and presented the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Guildman Altisen]";
	mes "By the heat of our furnaces and the pride of Einbroch, I name you a Blacksmith!";
	next;

	delitem 50008, 1;
	jobchange Job_Blacksmith;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Guildman Altisen]";
	mes "Arise as a ^0000FFBlacksmith^000000!";
	mes "Forge legendary weapons, enhance your allies' strength, and let your hammer echo across the world!";
	next;
	mes "[Guildman Altisen]";
	mes "Ah! Before you head to the forge, here's an interesting piece of news.";
	mes "Travelers from the capital report that a distinguished ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Guildman Altisen]";
	mes "He is searching for adventurers of proven valor to aid him in a crucial matter.";
	mes "They say he wanders near the library and sanctuary district of Prontera... Seek him out if you seek new horizons!";
	close;
}
