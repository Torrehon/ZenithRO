prt_church,16,41,4	script	High Bishop#prst	60,{
	mes "[High Bishop Paul]";
	if (Upper == 1) {
		mes "May the divine light envelope your soul. A Transcendent servant of the holy light stands before me.";
		next;
		mes "[High Bishop Paul]";
		mes "Go forth and shine as a beacon of hope in this dark world.";
		close;
	}
	if (BaseJob != Job_Acolyte) {
		if (BaseJob == Job_Priest) {
			mes "Blessings upon you, holy brother. May the divine sanctuary protect you and your companions always.";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to the Grand Sanctuary of Prontera! To devote yourself to holy magic, first become an Acolyte here in the church.";
			close;
		}
		else {
			mes "Peace be with you. The doors of the Prontera Sanctuary are always open to all weary travelers.";
			close;
		}
	}

	mes "Welcome to the sacred altar of the Prontera Sanctuary.";
	mes "Here, under the watchful gaze of the heavens, our order devotes itself to healing the sick, protecting the vulnerable, and exorcising evil across Rune-Midgard.";
	next;
	mes "[High Bishop Paul]";
	mes "To be ordained as a ^0000FFPriest^000000, an Acolyte must demonstrate unwavering faith, pure devotion, and holy wisdom.";
	next;
	mes "[High Bishop Paul]";
	mes "The sanctuary council requires two conditions for your holy ordination:";
	mes "1. Reach ^0000FFJob Level 50^000000 as an Acolyte.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), confirming your holy prestige and service to the realm.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[High Bishop Paul]";
		mes "Let us check your spiritual readiness...";
		next;
		if (JobLevel < 50) {
			mes "[High Bishop Paul]";
			mes "Your Job Level is currently " + JobLevel + ". Pray and serve the faithful until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[High Bishop Paul]";
			mes "You do not possess a ^FF0000Proof of Valor^000000. Obtain this testament of faith and courage from the realm's associations before receiving your vows.";
		}
		close;
	}

	mes "[High Bishop Paul]";
	mes "Glory to the divine light! You have reached Job Level 50 and presented the sacred ^FF0000Proof of Valor^000000!";
	next;
	mes "[High Bishop Paul]";
	mes "By the holy authority of the Sanctuary of Prontera, I ordain you as a Priest of the Realm!";
	next;

	delitem 50008, 1;
	jobchange Job_Priest;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[High Bishop Paul]";
	mes "Arise as a ^0000FFPriest of Light^000000!";
	mes "May your prayers heal the wounded, your Magnificat restore energy, and your Turn Undead banish darkness!";
	next;
	mes "[High Bishop Paul]";
	mes "Before you leave our holy altar, there is a matter of spiritual interest you should know.";
	mes "A learned ^0000FFScholar^000000 has recently arrived right here in ^0000FFProntera^000000.";
	next;
	mes "[High Bishop Paul]";
	mes "He has been searching for faithful heroes of proven valor to assist in a matter of profound spiritual importance.";
	mes "He wanders near our church sanctuary and the library district... Seek him out when you are ready to answer the call!";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}
