gef_tower,111,37,4	script	Wizard Guildsman#wiz	70,{
	mes "[Wizard Catherine]";
	if (Upper == 1) {
		mes "The arcane weave vibrates with incredible intensity around you! A Transcendent Archmage stands in our tower.";
		next;
		mes "[Wizard Catherine]";
		mes "May the elemental currents ever obey your command.";
		close;
	}
	if (BaseJob != Job_Mage) {
		if (BaseJob == Job_Wizard) {
			mes "Greetings, fellow Wizard! May your spells freeze, incinerate, and electrify the enemies of Geffen!";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to the Magic Tower of Geffen! To master high sorcery, train first as a Mage here in Geffen.";
			close;
		}
		else {
			mes "Greetings, traveler. Feel free to study the ancient arcane tomes inside Geffen Tower.";
			close;
		}
	}

	mes "Welcome to the top floor of Geffen's Magic Tower!";
	mes "For centuries, the Wizard Guild of Geffen has harnessed the destructive elemental forces of nature—fire, ice, lightning, and earth—to protect our realm.";
	next;
	mes "[Wizard Catherine]";
	mes "To earn the title of ^0000FFWizard^000000, a Mage must possess immense mana capacity, absolute elemental focus, and high arcane discipline.";
	next;
	mes "[Wizard Catherine]";
	mes "Our council of archmagi requires two credentials for your induction:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Mage.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige and magical prowess.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Wizard Catherine]";
		mes "Let me measure your mana reserves and credentials...";
		next;
		if (JobLevel < 50) {
			mes "[Wizard Catherine]";
			mes "Your Job Level is currently " + JobLevel + ". Continue your magic studies until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Wizard Catherine]";
			mes "You do not have a ^FF0000Proof of Valor^000000. Obtain this testament of merit from the realm's associations before ascending to high sorcery.";
		}
		close;
	}

	mes "[Wizard Catherine]";
	mes "Marvelous! You have attained Job Level 50 and presented the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Wizard Catherine]";
	mes "By the arcane flame of Geffen and the wisdom of the high tower, I proclaim you a Wizard!";
	next;

	delitem 50008, 1;
	jobchange Job_Wizard;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Wizard Catherine]";
	mes "Arise as a ^0000FFWizard^000000!";
	mes "Unleash devastating Storm Gusts, summon Meteors, and command the elements at your command!";
	next;
	mes "[Wizard Catherine]";
	mes "Before you descend from our magic tower, an intriguing magical report reached us from the capital.";
	mes "An erudite ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Wizard Catherine]";
	mes "He is inquiring about accomplished adventurers who hold genuine proof of valor to aid him in a crucial endeavor.";
	mes "He lingers near the library and sanctuary district of Prontera... Pay him a visit if you seek further knowledge!";
	close;
}
