hu_in01,386,373,4	script	Hunter Guildsman#hnt	727,{
	mes "[Hunter Sherin]";
	if (Upper == 1) {
		mes "The wind carries your step, yet you make no sound. A Transcendent hunter of the wild!";
		next;
		mes "[Hunter Sherin]";
		mes "May your arrows strike true and your falcon soar high above the clouds.";
		close;
	}
	if (BaseJob != Job_Archer) {
		if (BaseJob == Job_Hunter) {
			mes "Greetings, master of the wilderness! May the forests grant you keen sight and swift prey.";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to the Hunter Guild in Hugel! To walk the path of the hunter, train first as an Archer in Payon.";
			close;
		}
		else {
			mes "Welcome to our guild! Breathe in the fresh mountain air of Hugel and respect nature.";
			close;
		}
	}

	mes "Welcome to the Hunter Guild of Hugel!";
	mes "Far from the noise of big cities, our guild members live in harmony with the wild, tracking quarry through dense forests and rugged peaks.";
	next;
	mes "[Hunter Sherin]";
	mes "To earn the rank of ^0000FFHunter^000000, an Archer must prove immaculate focus, survival instinct, and mastery over nature.";
	next;
	mes "[Hunter Sherin]";
	mes "Our guild leaders require two credentials:";
	mes "1. Reach ^0000FFJob Level 50^000000 as an Archer.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), demonstrating your prestige across Rune-Midgard.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Hunter Sherin]";
		mes "Let me check your tracking record...";
		next;
		if (JobLevel < 50) {
			mes "[Hunter Sherin]";
			mes "Your Job Level is currently " + JobLevel + ". Hone your archery in the wilderness until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Hunter Sherin]";
			mes "You do not have a ^FF0000Proof of Valor^000000. Bring this token of merit from the realm's associations to complete your induction.";
		}
		close;
	}

	mes "[Hunter Sherin]";
	mes "Splendid! You have achieved Job Level 50 and brought forth the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Hunter Sherin]";
	mes "By the wind of Hugel and the spirit of the wild falcon, I welcome you as a Hunter!";
	next;

	delitem 50008, 1;
	jobchange Job_Hunter;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Hunter Sherin]";
	mes "Rise as a ^0000FFHunter^000000!";
	mes "Set your deadly traps, bond with your falcon, and let no target escape your sight!";
	next;
	mes "[Hunter Sherin]";
	mes "Before you venture into the wild, the wind carries a rumour from the capital.";
	mes "An intriguing ^0000FFScholar^000000 has been seen inquiring in ^0000FFProntera^000000.";
	next;
	mes "[Hunter Sherin]";
	mes "He is seeking courageous warriors who possess true valor to assist him in an extraordinary quest.";
	mes "He lingers around Prontera's quiet library district... Track him down if your journey takes you there!";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}
