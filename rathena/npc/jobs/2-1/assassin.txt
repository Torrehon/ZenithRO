in_moc_16,19,33,1	script	Guildsman#asn	55,{
	mes "[Guildman Kai]";
	if (Upper == 1) {
		mes "Shh... The shadows bend around your presence. A Transcendent master of the unseen walk.";
		next;
		mes "[Guildman Kai]";
		mes "May your blades stay silent and your venom strike true.";
		close;
	}
	if (BaseJob != Job_Thief) {
		if (BaseJob == Job_Assassin) {
			mes "Greetings, shadow brother. Walk quietly and never let your target see the flash of your Katar.";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "A novice in the desert? If you seek the art of the blade, become a Thief in Morroc first.";
			close;
		}
		else {
			mes "You are in the sanctum of the Assassin Guild. Watch your step, for the shadows here bite back.";
			close;
		}
	}

	mes "Welcome to the hidden enclave of the Assassin Guild.";
	mes "Deep beneath the sands of Sograt Desert, our brotherhood has perfected the lethal art of assassination, stealth, and venom for centuries.";
	next;
	mes "[Guildman Kai]";
	mes "To earn the privilege of wielding dual daggers and the lethal Katar as an ^0000FFAssassin^000000, our elders require proof of unmatched prowess.";
	next;
	mes "[Guildman Kai]";
	mes "You must satisfy two strict conditions:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Thief.";
	mes "2. Bring a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your lethality and discipline to the high council.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Guildman Kai]";
		mes "Let me inspect your credentials...";
		next;
		if (JobLevel < 50) {
			mes "[Guildman Kai]";
			mes "Your Job Level is currently " + JobLevel + ". Hone your speed until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Guildman Kai]";
			mes "You lack a ^FF0000Proof of Valor^000000. Earn this mark of prestige from the kingdom's associations and return to the shadows.";
		}
		close;
	}

	mes "[Guildman Kai]";
	mes "Impressive! You possess both the Job Level 50 mastery and the sacred ^FF0000Proof of Valor^000000!";
	next;
	mes "[Guildman Kai]";
	mes "The shadow elders accept your allegiance. Welcome to the Assassin Guild.";
	next;

	delitem 50008, 1;
	jobchange Job_Assassin;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Guildman Kai]";
	mes "Rise as an ^0000FFAssassin^000000!";
	mes "Strike from the dark, let your Katar dance, and remain unseen!";
	next;
	mes "[Guildman Kai]";
	mes "Before you return to the shadows, listen closely to a rumor from our informants.";
	mes "A mysterious ^0000FFScholar^000000 has been spotted inquiring in ^0000FFProntera^000000.";
	next;
	mes "[Guildman Kai]";
	mes "He seeks heroes of proven valor for an endeavor of great spiritual significance.";
	mes "He lingers near Prontera's library and sanctuary district... Keep your eyes open if your path leads to the capital!";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}
