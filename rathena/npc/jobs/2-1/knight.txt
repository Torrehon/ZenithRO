// Quest Principal (Capitán Herman)

prt_in,88,101,4	script	Chivalry Captain#knt	56,{
	mes "[Captain Herman]";
	
	if (BaseJob == Job_Knight) {
		mes "Welcome to the Chivalry, brother.";
		mes "I hope you continue to protect the weak with the honor that characterizes you.";
		close;
	}
	
	if (BaseJob != Job_Swordman || JobLevel < 50) {
		mes "We only accept Swordmen who have mastered their skills and reached Job Level 50.";
		mes "Return when you have reached your maximum potential.";
		close;
	}

	// Fase 0: Lore y Proof of Valor
	if (KNIGHT_Q == 0) {
		mes "Halt right there, stranger!";
		mes "The illustrious Prontera Chivalry is not a daycare for novice adventurers who barely know how to hold a sword.";
		next;
		mes "[Captain Herman]";
		mes "If you think you have what it takes, you must first prove that you are not a complete nobody.";
		mes "Return when you have a ^236B8EProof of Valor^000000. Until then, do not waste my time.";
		next;
		
		if (countitem(50008) >= 1) { 
			mes "[Captain Herman]";
			mes "Wait a moment... Is that a real Proof of Valor?";
			mes "My apologies, " + strcharinfo(0) + ". I didn't realize we were in the presence of someone of your renown.";
			next;
			delitem 50008,1;
			set KNIGHT_Q, 1;
			mes "[Captain Herman]";
			mes "Consider the doors of the Chivalry open to you.";
			mes "However, our rules demand a martial test. Talk to me when you are ready to enter the combat arena.";
			close;
		} else {
			mes "[Captain Herman]";
			mes "As I said. No badge, no deal. Good day!";
			close;
		}
	}

	// Fase 1: Entrada a la arena
	if (KNIGHT_Q == 1) {
		mes "Are you ready for your combat test?";
		mes "You will face formidable creatures. There will be no mercy.";
		next;
		if (select("Yes, take me to the arena.:I need to prepare more.") == 1) {
			mes "[Captain Herman]";
			mes "May the gods guide your sword!";
			close2;
			warp "job_knt",89,101; 
			end;
		}
		mes "[Captain Herman]";
		mes "Wise decision. Go sharpen your weapon.";
		close;
	}

	// Fase 2: Prueba superada, cambio de Job
	if (KNIGHT_Q == 2) {
		mes "You did it! You survived the combat test and demonstrated unyielding strength.";
		next;
		mes "[Captain Herman]";
		mes "I have no doubts about you.";
		mes "I, Herman Phon Efesirsus, officially name you a Knight of Prontera.";
		next;
		
		jobchange Job_Knight;
		set KNIGHT_Q, 0; 
		callfunc "F_ClearJobVar"; 
		getitem 656,7; 
		
		mes "[Captain Herman]";
		mes "Here are some potions for your future battles.";
		mes "Fight with honor and never surrender!";
		close;
	}
}

OnInit:
	waitingroom "Waiting Room",20,"Arena_Controller#knt::OnStartArena",1;
	enablewaitingroomevent;
	end;

OnStartArena:
	disablewaitingroomevent; 
	warpwaitingpc "job_knt",43,146; 
	donpcevent "KnightCombat::OnEnable";
	end;

OnFinish:
	enablewaitingroomevent; 
	end;
}

//==========================================================
// Lógica de la Arena (Oleadas, Tiempos y Reseteo por Muerte)
//==========================================================
-	script	KnightCombat	-1,{
OnInit:
	disablenpc "KnightCombat";
	end;

OnEnable:
	enablenpc "KnightCombat";
	set .Wave, 0;
	set .MyMobs, 0;
	initnpctimer; // Inicia el reloj (Única vez que se llama en toda la prueba)
	end;

OnDisable:
	killmonster "job_knt", "KnightCombat::OnMobDead";
	disablenpc "KnightCombat";
	end;

// --- Detección de Muerte o Desconexión del Jugador ---
OnPCDieEvent:
	// Solo actuamos si el jugador muere DENTRO de la arena del Knight
	if (strcharinfo(3) == "job_knt") {
		stopnpctimer;
		mapannounce "job_knt", "The candidate has fallen! The arena is resetting.", bc_red|bc_map;
		sleep2 3000; // Damos 3 segundos para que el jugador lea el mensaje
		percentheal 100, 100; // Lo resucitamos automáticamente
		warp "prt_in", 80, 100; // Lo expulsamos a Prontera Chivalry
		donpcevent "KnightCombat::OnDisable";
		donpcevent "Arena_Controller#knt::OnFinish";
	}
	end;

OnPCLogoutEvent:
	// Si el jugador se desconecta, liberamos la sala instantáneamente
	if (strcharinfo(3) == "job_knt") {
		stopnpctimer;
		donpcevent "KnightCombat::OnDisable";
		donpcevent "Arena_Controller#knt::OnFinish";
	}
	end;

// --- Intro ---
OnTimer1000:
	mapannounce "job_knt", "Welcome to the Combat Arena! Prepare your gear.", bc_map;
	end;
OnTimer5000:
	mapannounce "job_knt", "Monsters will be unleashed in 5 seconds...", bc_map;
	end;
OnTimer10000:
	// Bloqueo de seguridad para que no invoque si la oleada ya avanzó
	if (.Wave != 0) end; 
	
	set .Wave, 1;
	set .MyMobs, 10; 
	
	monster "job_knt",40,140,"Orc Warrior",1023,1,"KnightCombat::OnMobDead";
	monster "job_knt",46,140,"Orc Warrior",1023,1,"KnightCombat::OnMobDead";
	monster "job_knt",43,146,"Orc Warrior",1023,1,"KnightCombat::OnMobDead";
	monster "job_knt",38,148,"Orc Warrior",1023,1,"KnightCombat::OnMobDead";
	monster "job_knt",48,148,"Orc Warrior",1023,1,"KnightCombat::OnMobDead";
	
	monster "job_knt",40,152,"Orc Lady",1273,1,"KnightCombat::OnMobDead";
	monster "job_knt",46,152,"Orc Lady",1273,1,"KnightCombat::OnMobDead";
	monster "job_knt",43,138,"Orc Lady",1273,1,"KnightCombat::OnMobDead";
	monster "job_knt",38,142,"Orc Lady",1273,1,"KnightCombat::OnMobDead";
	monster "job_knt",48,142,"Orc Lady",1273,1,"KnightCombat::OnMobDead";
	
	mapannounce "job_knt", "Wave 1: Defeat the Orc swarm!", bc_map;
	end;

// --- Control de Muerte de Monstruos ---
OnMobDead:
	set .MyMobs, .MyMobs - 1;
	
	if (.MyMobs <= 0) {
		if (.Wave == 1) {
			set .Wave, 2;
			mapannounce "job_knt", "Wave 1 cleared! Get ready for the final challenge...", bc_map;
			
			// Usamos sleep2 para no perder el rastro del jugador
			sleep2 4000; 
			
			set .MyMobs, 4; 
			mapannounce "job_knt", "Wave 2: High Orcs and Orc Archers have spawned!", bc_map;
			
			monster "job_knt",36,147,"High Orc",1213,1,"KnightCombat::OnMobDead";
			monster "job_knt",36,145,"High Orc",1213,1,"KnightCombat::OnMobDead";
			monster "job_knt",50,147,"Orc Archer",1189,1,"KnightCombat::OnMobDead";
			monster "job_knt",50,145,"Orc Archer",1189,1,"KnightCombat::OnMobDead";
			
		} else if (.Wave == 2) {
			stopnpctimer; // Frenamos el temporizador general de la arena
			mapannounce "job_knt", "Congratulations! You have cleared the trial!", bc_map;
			mapannounce "job_knt", "You have 10 seconds to collect your loot before teleporting.", bc_map;
			set KNIGHT_Q, 2;
			
			// Usamos sleep2 de nuevo para mantener al jugador anclado al script
			sleep2 10000; 
			
			// Ahora el warp sí reconocerá a tu jugador
			warp "prt_in",80,100; 
			donpcevent "KnightCombat::OnDisable";
			donpcevent "Arena_Controller#knt::OnFinish";
		}
	}
	end;

// --- Tiempo Límite de la Arena (5 Minutos en total = 300,000 ms) ---
OnTimer300000: 
	stopnpctimer;
	mapannounce "job_knt", "Time is up! You have failed the test.", bc_map;
	
	// 'mapwarp' barre la sala entera y expulsa a todos. Ideal para OnTimer que no tiene un jugador anclado.
	mapwarp "job_knt", "prt_in", 80, 100; 
	
	donpcevent "KnightCombat::OnDisable";
	donpcevent "Arena_Controller#knt::OnFinish";
	end;
}
