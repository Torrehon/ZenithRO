prt_in,88,101,4	script	Chivalry Captain#knt	56,{
	mes "[Captain Herman]";
	if (Upper == 1) {
		mes "Greetings, noble hero. The radiant aura of a Transcendent warrior shines brightly from within you.";
		next;
		mes "[Captain Herman]";
		mes "May the divine grace of King Tristan III and the spirit of chivalry guide your blade forever.";
		close;
	}
	if (BaseJob != Job_Swordman) {
		if (BaseJob == Job_Knight) {
			mes "Hail, brother in arms! As Knights of the Prontera Chivalry, we stand as the unbreakable shield of Rune-Midgard.";
			mes "Keep your sword sharp and your honor untarnished!";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to the Prontera Chivalry!";
			mes "If you aspire to walk the path of the Knight, you must first train as a proud Swordman in Izlude.";
			close;
		}
		else {
			mes "Greetings! The Prontera Chivalry welcomes all allies of the kingdom. Honor and strength be with you.";
			close;
		}
	}

	mes "Welcome to the headquarters of the Prontera Chivalry, warrior.";
	mes "For generations, our order has stood as the supreme vanguard of the Kingdom of Rune-Midgard, protecting our land from darkness.";
	next;
	mes "[Captain Herman]";
	mes "To earn the title of ^0000FFKnight^000000, a Swordman must prove complete mastery over combat and unwavering valor.";
	next;
	mes "[Captain Herman]";
	mes "Our high council requires two conditions for your Knighthood induction:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Swordman.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), the sacred mark granted by the realm's associations of prestige.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Captain Herman]";
		mes "Let us check your credentials...";
		next;
		if (JobLevel < 50) {
			mes "[Captain Herman]";
			mes "Your current Job Level is " + JobLevel + ". You must train further until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Captain Herman]";
			mes "You do not possess a ^FF0000Proof of Valor^000000. Obtain this testament of courage from the prestige associations of the realm and return to me.";
		}
		close;
	}

	mes "[Captain Herman]";
	mes "Incredible! You have reached Job Level 50 and brought forth the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Captain Herman]";
	mes "By the power vested in me by the Prontera Chivalry and the Kingdom of Rune-Midgard, I accept your oath.";
	next;

	delitem 50008, 1;
	jobchange Job_Knight;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Captain Herman]";
	mes "Arise, ^0000FFKnight of Rune-Midgard^000000!";
	mes "May your courage never falter, your spear strike true, and your shield protect the innocent!";
	next;
	mes "[Captain Herman]";
	mes "Before you depart to serve the kingdom, take heed of this news.";
	mes "Word has reached the Chivalry that an intriguing ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Captain Herman]";
	mes "He is seeking brave warriors of proven valor to assist in a matter of great spiritual importance.";
	mes "He was last seen wandering near the quiet library and sanctuary district of Prontera... Seek him out if you are ready for a higher calling!";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}
