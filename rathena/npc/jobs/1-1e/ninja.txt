alberta,30,65,3	script	Akagi#nj	730,{
	mes "[Akagi]";
	mes "Greetings, traveler. If you seek the secret shadows of the Ninja Guild in Amatsu, Kuuga Gai awaits inside the secret enclave.";
	next;
	if (select("Warp to Ninja Enclave:Stay in Alberta") == 1) {
		mes "[Akagi]";
		mes "May the shadows conceal your footsteps...";
		close2;
		warp "que_ng",30,60;
		end;
	}
	mes "[Akagi]";
	mes "Farewell.";
	close;
}

que_ng,30,65,3	script	Kuuga Gai#nq	730,{
	mes "[Kuuga Gai]";
	if (Upper == 2) {
		mes "Incredible spirit! A Transcendent shadow warrior stands before me.";
		close;
	}
	if (BaseJob != Job_Novice) {
		if (BaseJob == Job_Ninja) {
			mes "Greetings, shadow brother! Let your Huuma Shuriken spin and your Cicada Skin Shed protect you from every strike.";
			close;
		}
		else {
			mes "Welcome to the Ninja Enclave. Walk quietly and respect the shadow clan.";
			close;
		}
	}

	mes "Welcome to the secret sanctum of the Ninja Guild!";
	mes "Hailing from the misty realm of Amatsu, our order combines martial ninjutsu, shadow mobility, lethal Huuma Shurikens, and elemental magic spells.";
	next;
	mes "[Kuuga Gai]";
	mes "To earn the headband of a ^0000FFNinja^000000, a Novice must demonstrate complete patience, stealth, and battle experience.";
	next;
	mes "[Kuuga Gai]";
	mes "Our clan requires two credentials:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Novice.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige across Rune-Midgard.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Kuuga Gai]";
		mes "Let me inspect your shadow credentials...";
		next;
		if (JobLevel < 50) {
			mes "[Kuuga Gai]";
			mes "Your Job Level is currently " + JobLevel + ". Hone your discipline until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Kuuga Gai]";
			mes "You do not have a ^FF0000Proof of Valor^000000. Obtain this testament of merit from the realm's associations before taking the shadow oath.";
		}
		close;
	}

	mes "[Kuuga Gai]";
	mes "Exemplary! You have achieved Job Level 50 and presented the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Kuuga Gai]";
	mes "By the shadow clan of Amatsu, I welcome you as a true Ninja!";
	next;

	delitem 50008, 1;
	jobchange Job_Ninja;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Kuuga Gai]";
	mes "Arise as a ^0000FFNinja^000000!";
	mes "Throw Huuma Shurikens, execute Shadow Jump, cast Cicada Skin Shed, and command Ninjutsu magic!";
	next;
	mes "[Kuuga Gai]";
	mes "Listen closely before you step into the world...";
	mes "Our informants report that a mysterious ^0000FFScholar^000000 has recently arrived in ^0000FFProntera^000000.";
	next;
	mes "[Kuuga Gai]";
	mes "He is actively seeking proven heroes who possess authentic valor for a matter of great spiritual importance.";
	mes "He lingers near the library and sanctuary district of Prontera... Seek him out if you are ready for a higher calling!";
	close;
}
