//= Gunslinger Job Quest (Simplified - Job 50 + Proof of Valor)
//===== Description: ========================================= 
//= Job Change quest for Gunslinger class.
//============================================================ 

que_ng,152,167,3	script	Master Miller#guns	901,{
	mes "[Master Miller]";
	if (Upper == 2) {
		mes "Well, aren't you an adorable warrior~";
		mes "Stay safe out there!";
		close;
	}
	if (BaseJob != Job_Novice) {
		if (BaseJob == Job_Gunslinger) {
			mes "Greetings, sharp-shooter! Keep your gunpowder dry, your trigger finger steady, and your aim true!";
			close;
		}
		else {
			mes "Welcome to the Gunslinger Guild of Einbroch! We train elite marksmen who wield firearms and explosives.";
			close;
		}
	}

	mes "Welcome to the Gunslinger Guild!";
	mes "While traditional archers use wood and string, Gunslingers command iron and gunpowder—mastering revolvers, rifles, shotguns, gatling guns, and grenade launchers to dominate the battlefield.";
	next;
	mes "[Master Miller]";
	mes "To earn the mark of a ^0000FFGunslinger^000000, a Novice must demonstrate absolute aim, steady hands, and battle experience.";
	next;
	mes "[Master Miller]";
	mes "Our guild requires two credentials:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Novice.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige across Rune-Midgard.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Master Miller]";
		mes "Let me inspect your marksmanship credentials...";
		next;
		if (JobLevel < 50) {
			mes "[Master Miller]";
			mes "Your Job Level is currently " + JobLevel + ". Hone your combat experience until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Master Miller]";
			mes "You do not possess a ^FF0000Proof of Valor^000000. Obtain this testament of courage from the realm's associations before receiving your firearm license.";
		}
		close;
	}

	mes "[Master Miller]";
	mes "Outstanding! You have reached Job Level 50 and presented the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Master Miller]";
	mes "By the gunpowder of Einbroch and the authority of the Gunslinger Guild, I name you a Gunslinger!";
	next;

	delitem 50008, 1;
	jobchange Job_Gunslinger;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Master Miller]";
	mes "Arise as a ^0000FFGunslinger^000000!";
	mes "Master Rapid Shower, Desperado, Gatling Fever, and let the thunder of your guns echo across the land!";
	next;
	mes "[Master Miller]";
	mes "Ah, before you head out, there is one more thing you should know...";
	mes "Word has reached our guild that an intriguing ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Master Miller]";
	mes "He has been inquiring about brave, seasoned adventurers who hold proof of great valor to assist him in an extraordinary endeavor.";
	mes "They say he lingers near the quiet sanctuary and library district of Prontera... Seek him out if you are ready for a new destiny!";
	close;
}
