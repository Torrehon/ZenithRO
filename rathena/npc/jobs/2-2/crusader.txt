-	script	Senior Crusader_	-1,{
	mes "[Senior Crusader Michael]";
	if (Upper == 1) {
		mes "The radiant holy aura of a Transcendent holy warrior shines from within you!";
		next;
		mes "[Senior Crusader Michael]";
		mes "May your holy shield never fall and your faith remain absolute.";
		close;
	}
	if (BaseJob != Job_Swordman) {
		if (BaseJob == Job_Crusader) {
			mes "Hail, holy brother in arms! Stand as the divine shield of Rune-Midgard and protect your companions with your life!";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to Prontera Castle! To walk the holy path of the Crusader, train first as a Swordman in Izlude.";
			close;
		}
		else {
			mes "Welcome to the Royal Guard hall. Honor and divine protection be with you.";
			close;
		}
	}

	mes "Welcome to the Royal Order of Crusaders in Prontera Castle!";
	mes "While Knights act as the spearhead of the kingdom, Crusaders are the divine bulwark—sworn holy warriors who channel divine power to protect the innocent and banish darkness.";
	next;
	mes "[Senior Crusader Michael]";
	mes "To earn the holy shield of a ^0000FFCrusader^000000, a Swordman must demonstrate unwavering faith, selflessness, and mastery over holy shield combat.";
	next;
	mes "[Senior Crusader Michael]";
	mes "Our holy order requires two credentials for your induction:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Swordman.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige across Rune-Midgard.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Senior Crusader Michael]";
		mes "Let me verify your holy readiness...";
		next;
		if (JobLevel < 50) {
			mes "[Senior Crusader Michael]";
			mes "Your Job Level is currently " + JobLevel + ". Hone your martial and spiritual strength until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Senior Crusader Michael]";
			mes "You do not possess a ^FF0000Proof of Valor^000000. Obtain this testament of courage from the kingdom's prestige associations to complete your induction.";
		}
		close;
	}

	mes "[Senior Crusader Michael]";
	mes "Glorious! You have attained Job Level 50 and brought forth the sacred ^FF0000Proof of Valor^000000!";
	next;
	mes "[Senior Crusader Michael]";
	mes "By the royal authority of Prontera Castle and the holy light, I ordain you a Crusader!";
	next;

	delitem 50008, 1;
	jobchange Job_Crusader;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Senior Crusader Michael]";
	mes "Arise as a ^0000FFCrusader^000000!";
	mes "Wield Grand Cross, cast Devotion to protect your allies, and stand as an unyielding holy fortress!";
	next;
	mes "[Senior Crusader Michael]";
	mes "Before you step outside castle walls, take note of an official report.";
	mes "A distinguished ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Senior Crusader Michael]";
	mes "He is seeking holy warriors and heroes of proven valor for a mission of great spiritual importance.";
	mes "He lingers near the library and sanctuary district of Prontera... Seek him out if your shield is ready!";
	close;
}
