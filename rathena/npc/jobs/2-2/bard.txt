//===== rAthena Script ======================================= 
//= Bard Job Quest (Simplified - Job 50 + Proof of Valor)
//===== Description: ========================================= 
//= Job Change quest for Bard class.
//============================================================ 

comodo,226,123,5	script	Wandering Bard#bard	741,{
	mes "[Wandering Bard Luluka]";
	if (Upper == 1) {
		mes "Ah! The melody of a Transcendent troubadour echoes in your soul!";
		next;
		mes "[Wandering Bard Luluka]";
		mes "May your songs ever inspire courage and bring joy to all who listen.";
		close;
	}
	if (Sex != SEX_MALE) {
		mes "[Wandering Bard Luluka]";
		mes "Ah, beautiful lady! The path of the Bard is reserved for male troubadours. If you seek to dance to the music of Comodo, speak with the Dancers nearby!";
		close;
	}
	if (BaseJob != Job_Archer) {
		if (BaseJob == Job_Bard) {
			mes "Greetings, brother Bard! Play your instrument proudly and let your Poem of Bragi echo across the battlefield!";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Greetings, young traveler! To sing the ballads of Rune-Midgard, train first as an Archer in Payon.";
			close;
		}
		else {
			mes "Welcome to romantic Comodo! Take a moment to listen to the ocean waves and the gentle melody of my lute.";
			close;
		}
	}

	mes "Greetings, traveler! I am Luluka, the Wandering Bard of Comodo.";
	mes "Music is more than mere entertainment—a Bard's song can heal broken hearts, inspire warriors to fight beyond their limits, and strike dissonance into the minds of enemies.";
	next;
	mes "[Wandering Bard Luluka]";
	mes "To earn the instrument of a ^0000FFBard^000000, a male Archer must demonstrate poetic spirit, passion for song, and true battle experience.";
	next;
	mes "[Wandering Bard Luluka]";
	mes "I require two credentials before I pass my musical legacy to you:";
	mes "1. Reach ^0000FFJob Level 50^000000 as an Archer.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige across the realm.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Wandering Bard Luluka]";
		mes "Let me listen to the melody of your spirit...";
		next;
		if (JobLevel < 50) {
			mes "[Wandering Bard Luluka]";
			mes "Your Job Level is currently " + JobLevel + ". Hone your archery and song until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Wandering Bard Luluka]";
			mes "You lack a ^FF0000Proof of Valor^000000. Bring this token of merit from the realm's associations so I may compose a ballad in your honor.";
		}
		close;
	}

	mes "[Wandering Bard Luluka]";
	mes "Magnificent! You have achieved Job Level 50 and brought forth the legendary ^FF0000Proof of Valor^000000!";
	next;
	mes "[Wandering Bard Luluka]";
	mes "By the moon of Comodo and the rhythm of the ocean, I welcome you to the brotherhood of Bards!";
	next;

	delitem 50008, 1;
	jobchange Job_Bard;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Wandering Bard Luluka]";
	mes "Arise as a ^0000FFBard^000000!";
	mes "Play your lute, chant the Poem of Bragi, lower cast times for your allies, and fill the world with music!";
	next;
	mes "[Wandering Bard Luluka]";
	mes "Before you sing your first ballad, the wind carries a rumor from the capital!";
	mes "A mysterious ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Wandering Bard Luluka]";
	mes "He seeks heroes of proven valor to assist in an extraordinary quest of spiritual import.";
	mes "He wanders near Prontera's quiet library district... Seek him out if you desire to star in a grand new song!";
	close;
}
