yuno_in02,38,61,5	script	Dean of the Academy#sa	743,{
	mes "[Dean Kayron]";
	if (Upper == 1) {
		mes "The intellectual magic surrounding you is extraordinary! A Transcendent Scholar of the Yuno Academy.";
		next;
		mes "[Dean Kayron]";
		mes "May your academic research illuminate the grand truths of Rune-Midgard.";
		close;
	}
	if (BaseJob != Job_Mage) {
		if (BaseJob == Job_Sage) {
			mes "Greetings, esteemed Scholar! May your Free Cast mobility and Dispel unravel all enemy magic!";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to the Yuno Royal Academy! To study scholarly magic, train first as a Mage in Geffen.";
			close;
		}
		else {
			mes "Welcome to the Yuno Academy of Magic! Knowledge is the greatest power in the world.";
			close;
		}
	}

	mes "Welcome to the Yuno Royal Academy of Magic!";
	mes "While Wizards focus on raw elemental destruction, Sages master the science of magic—moving freely while casting spell formulas, disrupting enemy incantations, and endowing elemental fields.";
	next;
	mes "[Dean Kayron]";
	mes "To earn the degree of a ^0000FFSage^000000, a Mage must demonstrate high intellectual discipline, spell casting adaptability, and scholarly merit.";
	next;
	mes "[Dean Kayron]";
	mes "Our academy board requires two credentials for your graduation:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Mage.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige across Rune-Midgard.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Dean Kayron]";
		mes "Let me check your academic record...";
		next;
		if (JobLevel < 50) {
			mes "[Dean Kayron]";
			mes "Your Job Level is currently " + JobLevel + ". Continue your magical studies until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Dean Kayron]";
			mes "You do not have a ^FF0000Proof of Valor^000000. Obtain this testament of merit from the realm's associations before receiving your Sage diploma.";
		}
		close;
	}

	mes "[Dean Kayron]";
	mes "Outstanding! You have achieved Job Level 50 and presented the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Dean Kayron]";
	mes "By the authority of the Yuno Royal Academy, I confer upon you the degree of Sage!";
	next;

	delitem 50008, 1;
	jobchange Job_Sage;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Dean Kayron]";
	mes "Arise as a ^0000FFSage^000000!";
	mes "Master Free Cast, cast Spell Breaker and Dispel against your foes, and endow the battlefield with elemental power!";
	next;
	mes "[Dean Kayron]";
	mes "Before you leave the academy halls, an academic colleague of ours sent word from the capital.";
	mes "A distinguished ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Dean Kayron]";
	mes "He is seeking fellow scholars and warriors of proven valor to assist in an extraordinary research endeavor.";
	mes "He lingers near the library and sanctuary district of Prontera... Seek him out to expand your knowledge!";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}
