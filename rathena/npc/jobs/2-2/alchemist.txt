alde_alche,27,185,5	script	Alchemist Guildsman#am	744,{
	mes "[Alchemist Parmel]";
	if (Upper == 1) {
		mes "Fascinating chemical aura! A Transcendent master of alchemy stands in our laboratory.";
		next;
		mes "[Alchemist Parmel]";
		mes "May your flasks never break and your homunculus grow strong and loyal.";
		close;
	}
	if (BaseJob != Job_Merchant) {
		if (BaseJob == Job_Alchemist) {
			mes "Greetings, fellow researcher! May your potion creations achieve 100% success and your chemical bombs destroy all foes!";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to the Alchemist Society of Aldebaran! To learn chemical synthesis, train first as a Merchant in Alberta.";
			close;
		}
		else {
			mes "Welcome to our laboratory! Please do not touch any unlabelled test tubes or volatile acids.";
			close;
		}
	}

	mes "Welcome to the Alchemist Guild of Aldebaran!";
	mes "Here, science meets magic as our researchers combine rare herbs, minerals, and chemical catalysts to create miracle potions, explosive compounds, and living Homunculi.";
	next;
	mes "[Alchemist Parmel]";
	mes "To earn the title of ^0000FFAlchemist^000000, a Merchant must demonstrate deep understanding of chemistry and commercial synthesis.";
	next;
	mes "[Alchemist Parmel]";
	mes "Our board of directors requires two credentials for your induction:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Merchant.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige across Rune-Midgard.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Alchemist Parmel]";
		mes "Let me review your research credentials...";
		next;
		if (JobLevel < 50) {
			mes "[Alchemist Parmel]";
			mes "Your Job Level is currently " + JobLevel + ". Continue trading and learning until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Alchemist Parmel]";
			mes "You do not have a ^FF0000Proof of Valor^000000. Obtain this token of merit from the realm's associations before receiving your laboratory license.";
		}
		close;
	}

	mes "[Alchemist Parmel]";
	mes "Splendid! You have achieved Job Level 50 and presented the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Alchemist Parmel]";
	mes "By the authority of the Aldebaran Alchemist Guild, I grant you your official Alchemist license!";
	next;

	delitem 50008, 1;
	jobchange Job_Alchemist;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Alchemist Parmel]";
	mes "Arise as an ^0000FFAlchemist^000000!";
	mes "Brew potent potions, hurl Acid Terror, summon deadly flora, and nurture your Homunculus!";
	next;
	mes "[Alchemist Parmel]";
	mes "Oh, before you begin your chemical experiments, here is a note of interest.";
	mes "A prominent ^0000FFScholar^000000 has recently arrived in ^0000FFProntera^000000.";
	next;
	mes "[Alchemist Parmel]";
	mes "He is seeking adventurers of proven valor to assist in a matter of great importance.";
	mes "He is said to be wandering near the library and sanctuary district of Prontera... Seek him out when you visit the capital!";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}
