monk_in,99,58,1	script	Sensei Moohae#mk	60,{
	mes "[Sensei Moohae]";
	if (Upper == 1) {
		mes "The spirit spheres surrounding your body resonate with extraordinary power! A Transcendent champion of St. Eleanor Abbey.";
		next;
		mes "[Sensei Moohae]";
		mes "May your fists strike true and your mind remain tranquil as steel.";
		close;
	}
	if (BaseJob != Job_Acolyte) {
		if (BaseJob == Job_Monk) {
			mes "Greetings, martial brother! Channel your Spirit Spheres, steel your mind, and let your Asura Strike crush evil!";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to St. Eleanor Abbey! To learn martial holy combat, train first as an Acolyte in Prontera Church.";
			close;
		}
		else {
			mes "Greetings, traveler. Meditate quietly within our monastery halls.";
			close;
		}
	}

	mes "Welcome to St. Eleanor Abbey!";
	mes "While Priests heal with divine prayers, Monks turn their very bodies into divine weapons—combining physical martial arts with spiritual energy to banish evil.";
	next;
	mes "[Sensei Moohae]";
	mes "To earn the holy beads of a ^0000FFMonk^000000, an Acolyte must demonstrate absolute physical discipline, spirit sphere mastery, and iron resolve.";
	next;
	mes "[Sensei Moohae]";
	mes "Our monastery council requires two credentials:";
	mes "1. Reach ^0000FFJob Level 50^000000 as an Acolyte.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige across Rune-Midgard.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Sensei Moohae]";
		mes "Let me evaluate your spiritual endurance...";
		next;
		if (JobLevel < 50) {
			mes "[Sensei Moohae]";
			mes "Your Job Level is currently " + JobLevel + ". Continue your martial meditation until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Sensei Moohae]";
			mes "You do not have a ^FF0000Proof of Valor^000000. Obtain this testament of merit from the realm's associations before taking your martial vows.";
		}
		close;
	}

	mes "[Sensei Moohae]";
	mes "Exemplary! You have achieved Job Level 50 and presented the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Sensei Moohae]";
	mes "By the sacred lineage of St. Eleanor Abbey, I ordain you a Monk of Holy Martial Arts!";
	next;

	delitem 50008, 1;
	jobchange Job_Monk;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Sensei Moohae]";
	mes "Arise as a ^0000FFMonk^000000!";
	mes "Summon your Spirit Spheres, activate Steel Body, and unleash the devastating power of Asura Strike!";
	next;
	mes "[Sensei Moohae]";
	mes "Before you leave our monastery, heed this message from our brethren.";
	mes "A learned ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Sensei Moohae]";
	mes "He is actively searching for martial heroes who hold authentic proof of valor for a matter of great spiritual import.";
	mes "He wanders near the library and sanctuary district of Prontera... Seek him out to fulfill your spiritual destiny!";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}
