in_rogue,363,122,4	script	Rogue Guildsman#rg	747,{
	mes "[Rogue Guildmaster Mark]";
	if (Upper == 1) {
		mes "Well, well... the shadows bend around a Transcendent master rogue!";
		next;
		mes "[Rogue Guildmaster Mark]";
		mes "May your daggers hit the blind spots and your Plagiarism copy the greatest secrets of the realm.";
		close;
	}
	if (BaseJob != Job_Thief) {
		if (BaseJob == Job_Rogue) {
			mes "Welcome back to the syndicate, fellow Rogue! Keep your pockets full and your eyes open.";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to the Rogue Enclave! To learn our sneaky tricks, train first as a Thief in Morroc.";
			close;
		}
		else {
			mes "You are in the headquarters of the Rogue Guild. Watch your coin pouch!";
			close;
		}
	}

	mes "Welcome to the hidden stronghold of the Rogue Guild!";
	mes "While Assassins focus purely on lethal stealth, Rogues master versatile survival—combining daggers, bow mastery, Mug, Strip gear, and the secret art of Plagiarism to copy enemy skills.";
	next;
	mes "[Rogue Guildmaster Mark]";
	mes "To earn the crest of a ^0000FFRogue^000000, a Thief must demonstrate cunning adaptability, street smarts, and high battle skill.";
	next;
	mes "[Rogue Guildmaster Mark]";
	mes "Our syndicate requires two credentials for your initiation:";
	mes "1. Reach ^0000FFJob Level 50^000000 as a Thief.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige across Rune-Midgard.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Rogue Guildmaster Mark]";
		mes "Let me check your street reputation...";
		next;
		if (JobLevel < 50) {
			mes "[Rogue Guildmaster Mark]";
			mes "Your Job Level is currently " + JobLevel + ". Hone your stealth and speed until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Rogue Guildmaster Mark]";
			mes "You do not possess a ^FF0000Proof of Valor^000000. Earn this mark of merit from the realm's associations before entering our syndicate.";
		}
		close;
	}

	mes "[Rogue Guildmaster Mark]";
	mes "Impressive! You have attained Job Level 50 and presented the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Rogue Guildmaster Mark]";
	mes "By the authority of the Rogue Syndicate, welcome to the guild!";
	next;

	delitem 50008, 1;
	jobchange Job_Rogue;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Rogue Guildmaster Mark]";
	mes "Arise as a ^0000FFRogue^000000!";
	mes "Master Bow and Dagger combat, use Plagiarism to copy exotic skills, and rule the underground!";
	next;
	mes "[Rogue Guildmaster Mark]";
	mes "Oh, one more piece of street intelligence before you disappear into the night.";
	mes "A high-profile ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Rogue Guildmaster Mark]";
	mes "He is asking around for adventurers of proven valor to aid him in a discreet, high-value endeavor.";
	mes "He lingers near Prontera's library and sanctuary district... Keep your eyes sharp if you head to the capital!";
	close;
}
