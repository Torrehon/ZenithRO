job_duncer,43,93,4	script	Aile#da	724,{
	mes "[Instructor Aile]";
	if (Upper == 1) {
		mes "The elegance of your rhythm is undeniable! A Transcendent prima donna stands on our stage.";
		next;
		mes "[Instructor Aile]";
		mes "May your dance move hearts and grant victory on every battlefield.";
		close;
	}
	if (Sex != SEX_FEMALE) {
		mes "[Instructor Aile]";
		mes "Ah, handsome traveler! The path of the Dancer is reserved for female archers. If you seek musical arts, visit the Wandering Bard in Comodo!";
		close;
	}
	if (BaseJob != Job_Archer) {
		if (BaseJob == Job_Dancer) {
			mes "Greetings, prima donna! Dance with grace, whip your enemies with rhythm, and inspire your allies!";
			close;
		}
		else if (BaseClass == Job_Novice) {
			mes "Welcome to the Comodo Dance Stage! To learn the art of dance, train first as an Archer in Payon.";
			close;
		}
		else {
			mes "Welcome to our stage! Listen to the beat of the drums and feel the rhythm of Comodo.";
			close;
		}
	}

	mes "Welcome to the grand stage of the Comodo Dance Guild!";
	mes "Dance is not merely performance—it is a weapon of beauty. A Dancer's rhythm can bewitch enemies, increase party SP efficiency, and grant immense battle buffs.";
	next;
	mes "[Instructor Aile]";
	mes "To earn the whip of a ^0000FFDancer^000000, a female Archer must demonstrate immaculate rhythm, agility, and true battle experience.";
	next;
	mes "[Instructor Aile]";
	mes "Our dance academy requires two credentials:";
	mes "1. Reach ^0000FFJob Level 50^000000 as an Archer.";
	mes "2. Present a ^FF0000Proof of Valor^000000 (Item ID 50008), proving your prestige across the realm.";
	next;

	if (JobLevel < 50 || countitem(50008) < 1) {
		mes "[Instructor Aile]";
		mes "Let me evaluate your rhythm and credentials...";
		next;
		if (JobLevel < 50) {
			mes "[Instructor Aile]";
			mes "Your Job Level is currently " + JobLevel + ". Hone your archery and rhythm until you reach ^0000FFJob Level 50^000000.";
		}
		if (countitem(50008) < 1) {
			mes "[Instructor Aile]";
			mes "You do not have a ^FF0000Proof of Valor^000000. Obtain this token of merit from the realm's associations before taking your place in the spotlight.";
		}
		close;
	}

	mes "[Instructor Aile]";
	mes "Bravissima! You have achieved Job Level 50 and presented the authentic ^FF0000Proof of Valor^000000!";
	next;
	mes "[Instructor Aile]";
	mes "By the applause of Comodo and the rhythm of the tide, I welcome you as a Prima Donna Dancer!";
	next;

	delitem 50008, 1;
	jobchange Job_Dancer;
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_CLASSCHANGE;

	mes "[Instructor Aile]";
	mes "Arise as a ^0000FFDancer^000000!";
	mes "Wield your whip, dance Service for You, captivate your enemies, and command the spotlight!";
	next;
	mes "[Instructor Aile]";
	mes "Before you take the stage, a traveler from the capital shared an interesting rumor.";
	mes "An enigmatic ^0000FFScholar^000000 has arrived in ^0000FFProntera^000000.";
	next;
	mes "[Instructor Aile]";
	mes "He is seeking brave adventurers of proven valor to assist in an endeavor of great spiritual importance.";
	mes "He wanders near the sanctuary and library district of Prontera... Seek him out if you are ready for a new act!";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}
