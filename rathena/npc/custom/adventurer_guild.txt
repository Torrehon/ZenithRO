OnBuyItem:
	// Find which shop was just used
	.@idx = inarray(.Shops$, strnpcinfo(0));
	if (.@idx == -1) {
		// Try matching by attached master
		for (.@s = 0; .@s < getarraysize(.Shops$); .@s++) {
			if (strnpcinfo(0) == .Shops$[.@s]) {
				.@idx = .@s;
				break;
			}
		}
	}
	if (.@idx == -1) {
		dispbottom "System Error: Shop mapping not resolved.";
		end;
	}

	.@coin_id = .Coins[.@idx];
	.@host_npc$ = .Hosts$[.@idx];
	.@size = getarraysize(@bought_nameid);

	// Calculate total cost using host NPC's Shop_Data
	for (.@i = 0; .@i < .@size; .@i++) {
		.@item_idx = inarray(getd(.@host_npc$ + "::.Shop_Data"), @bought_nameid[.@i]);
		if (.@item_idx == -1) {
			dispbottom "System Error: Item not found in the supply line.";
			end;
		}
		.@total_cost += (getd(.@host_npc$ + "::.Shop_Data[" + (.@item_idx + 1) + "]") * @bought_quantity[.@i]);
	}

	if (.@total_cost > countitem(.@coin_id)) {
		mes "[^0055FFGuild Quartermaster^000000]";
		mes "I am afraid you lack the necessary " + .Cities$[.@idx] + " Coins for these supplies.";
		mes "Total required: " + .@total_cost + " Coins.";
		close;
	}

	delitem .@coin_id, .@total_cost;

	for (.@i = 0; .@i < .@size; .@i++) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
	}

	mes "[^0055FFGuild Quartermaster^000000]";
	mes "Requisition approved! May these provisions serve you well in the field.";
	mes "Safe travels, hunter!";
	specialeffect2 EF_MONEY;
	close;

OnInit:
	setarray .Cities$[0],
		"Prontera", "Payon", "Alberta", "Geffen", "Morroc", "Aldebaran", "Comodo", "Umbala",
		"Yuno", "Einbroch", "Hugel", "Niflheim", "Lutie";

// Wayfarer Master - Dungeon Waypoint Network
// Location: iz_ac01 (103, 44) | Facing: South (6) | Sprite: 448
// Unlocks dungeon entry waypoints permanently per account using coins
// =============================================================
iz_ac01,103,44,6	script	Wayfarer Master	448,{
	mes "[^0055FFWayfarer Master^000000]";
	mes "Greetings, adventurer. I am the Grand Cartographer and Wayfarer Master of the Agency.";
	mes "I map dangerous labyrinths across Midgard and grant permanent travel licenses to qualified hunters.";
	mes "Once unlocked, ^00AA00any character on your account^000000 can request teleportation from our field Wayfarer Warpers.";
	next;

	// Automatic migration from single-variable bitmask if present
	if (#WAYFARER_DUNGEONS) {
		#WAYFARER_DUNGEONS_0 = #WAYFARER_DUNGEONS_0 | #WAYFARER_DUNGEONS;
		#WAYFARER_DUNGEONS = 0;
	}

	if (getarraysize($@WF_Dun_Name$) == 0)
		callsub OnInit;

	while (1) {
		mes "[^0055FFWayfarer Master^000000]";
		mes "Select a contract tier to inspect available dungeon waypoints, or view your unlocked destinations:";
		next;

		set .@selTier, select(
			"^00AA00Tier D Waypoints (Novice / 25 Coins)^000000",
			"^0055FFTier C Waypoints (Intermediate / 50 Coins)^000000",
			"^FF8800Tier B Waypoints (Advanced / 100 Coins)^000000",
			"^FF0000Tier A Waypoints (Endgame / 150 Coins)^000000",
			"^AA00AAInspect Account Waypoints^000000",
			"Close"
		);

		if (.@selTier == 6) {
			mes "[^0055FFWayfarer Master^000000]";
			mes "May the winds guide your footsteps.";
			close;
		}

		// -----------------------------------------------------
		// Option 5: Inspect Account Waypoints
		// -----------------------------------------------------
		if (.@selTier == 5) {
			callfunc "F_WF_InspectDungeons";
			continue;
		}

		// -----------------------------------------------------
		// Options 1-4: Unlock by Tier
		// -----------------------------------------------------
		.@tier = .@selTier; // 1: D, 2: C, 3: B, 4: A
		deletearray .@tier_indices[0];
		.@tier_count = 0;
		.@menu$ = "";

		for (.@i = 0; .@i < $@WF_TotalDuns; .@i++) {
			if ($@WF_Dun_Tier[.@i] == .@tier) {
				.@tier_indices[.@tier_count] = .@i;
				.@tier_count++;
				if (getd("#WAYFARER_DUNGEONS_" + (.@i / 30)) & (1 << (.@i % 30))) {
					.@menu$ = .@menu$ + "^00AA00" + $@WF_Dun_Name$[.@i] + " (Unlocked)^000000:";
				} else {
					.@menu$ = .@menu$ + "^FF0000" + $@WF_Dun_Name$[.@i] + " (" + $@WF_Dun_Cost[.@i] + " " + $@WF_Dun_City$[.@i] + " Coins)^000000:";
				}
			}
		}
		.@menu$ = .@menu$ + "Back";

		mes "[^0055FFWayfarer Master^000000]";
		mes "Select a dungeon to review or unlock:";
		next;

		.@chosen_opt = select(.@menu$) - 1;
		if (.@chosen_opt == .@tier_count) continue;

		.@dun_idx = .@tier_indices[.@chosen_opt];

		// Check if already unlocked (using chunked 30-bit storage)
		if (getd("#WAYFARER_DUNGEONS_" + (.@dun_idx / 30)) & (1 << (.@dun_idx % 30))) {
			mes "[^0055FFWayfarer Master^000000]";
			mes "The ^0055FF" + $@WF_Dun_Name$[.@dun_idx] + "^000000 waypoint is already unlocked for your account!";
			mes "Speak with any ^0055FFWayfarer Warper^000000 in Midgard cities whenever you wish to teleport to its entrance.";
			next;
			continue;
		}

		// Unlock Dialogue & Cost Confirmation
		.@coin_id = $@WF_Dun_Coin[.@dun_idx];
		.@cost = $@WF_Dun_Cost[.@dun_idx];
		.@coin_name$ = $@WF_Dun_City$[.@dun_idx] + " Coin";

		mes "=== ^0055FFWaypoint Licensing^000000 ===";
		mes "Dungeon: ^FF0000" + $@WF_Dun_Name$[.@dun_idx] + "^000000";
		mes "Destination: ^777777" + $@WF_Dun_Map$[.@dun_idx] + " (" + $@WF_Dun_X[.@dun_idx] + ", " + $@WF_Dun_Y[.@dun_idx] + ")^000000";
		mes "Required License Fee: ^00AA00" + .@cost + "x " + .@coin_name$ + "^000000";
		mes "You currently hold: ^0055FF" + countitem(.@coin_id) + "x " + .@coin_name$ + "^000000";
		mes " ";
		mes "Would you like to permanently unlock this waypoint for your account?";
		next;

		if (select("Unlock Waypoint:Decline") == 2) {
			mes "[^0055FFWayfarer Master^000000]";
			mes "Let me know when you are ready to expand your map.";
			next;
			continue;
		}

		if (countitem(.@coin_id) < .@cost) {
			mes "[^0055FFWayfarer Master^000000]";
			mes "You do not have enough ^0055FF" + .@coin_name$ + "s^000000.";
			mes "You need ^FF0000" + .@cost + "^000000 coins, but only possess ^FF0000" + countitem(.@coin_id) + "^000000.";
			next;
			continue;
		}

		delitem .@coin_id, .@cost;
		.@chunk = .@dun_idx / 30;
		.@bit = 1 << (.@dun_idx % 30);
		setd "#WAYFARER_DUNGEONS_" + .@chunk, getd("#WAYFARER_DUNGEONS_" + .@chunk) | .@bit;

		specialeffect2 EF_INCAGILITY;
		specialeffect2 EF_COIN;
		mes "[^0055FFWayfarer Master^000000]";
		mes "License approved! The ^0055FF" + $@WF_Dun_Name$[.@dun_idx] + "^000000 waypoint is now ^00AA00permanently unlocked^000000 for your entire account!";
		mes "All ^0055FFWayfarer Warpers^000000 in Midgard cities can now transport your characters directly to the entrance.";
		next;
		continue;
	}

// =============================================================
// Function: F_WF_InspectDungeons
// Clean, paginated account waypoints overview for Wayfarer NPCs
// =============================================================
function	script	F_WF_InspectDungeons	{
	while (1) {
		// Calculate stats per tier
		.@unlocked_total = 0;
		deletearray .@tier_unlocked[0];
		deletearray .@tier_total[0];

		for (.@i = 0; .@i < $@WF_TotalDuns; .@i++) {
			.@t = $@WF_Dun_Tier[.@i];
			.@tier_total[.@t]++;
			if (getd("#WAYFARER_DUNGEONS_" + (.@i / 30)) & (1 << (.@i % 30))) {
				.@tier_unlocked[.@t]++;
				.@unlocked_total++;
			}
		}

		mes "=== ^0055FFWayfarer Network Status^000000 ===";
		mes "Account Progress: ^00AA00" + .@unlocked_total + " / " + $@WF_TotalDuns + "^000000 Dungeons";
		mes "Wayfarer Stones: ^0055FF" + countitem(50001) + "x Stones^000000";
		mes "--------------------------------";
		mes "^00AA00Tier D (Novice / 1 Stone):^000000 " + .@tier_unlocked[1] + " / " + .@tier_total[1];
		mes "^0055FFTier C (Standard / 2 Stones):^000000 " + .@tier_unlocked[2] + " / " + .@tier_total[2];
		mes "^FF8800Tier B (Advanced / 3 Stones):^000000 " + .@tier_unlocked[3] + " / " + .@tier_total[3];
		mes "^FF0000Tier A (Endgame / 4 Stones):^000000 " + .@tier_unlocked[4] + " / " + .@tier_total[4];
		next;

		.@sel = select(
			"^00AA00View Tier D Waypoints (" + .@tier_unlocked[1] + "/" + .@tier_total[1] + ")^000000",
			"^0055FFView Tier C Waypoints (" + .@tier_unlocked[2] + "/" + .@tier_total[2] + ")^000000",
			"^FF8800View Tier B Waypoints (" + .@tier_unlocked[3] + "/" + .@tier_total[3] + ")^000000",
			"^FF0000View Tier A Waypoints (" + .@tier_unlocked[4] + "/" + .@tier_total[4] + ")^000000",
			"Browse All Tiers (Page by Page)",
			"Return to Previous Menu"
		);

		if (.@sel == 6) {
			return;
		}

		if (.@sel <= 4) {
			.@view_tier = .@sel;
			.@tier_title$ = ((.@view_tier == 1) ? "^00AA00Tier D (Novice / 1 Stone)^000000" :
			                 (.@view_tier == 2) ? "^0055FFTier C (Standard / 2 Stones)^000000" :
			                 (.@view_tier == 3) ? "^FF8800Tier B (Advanced / 3 Stones)^000000" :
			                                      "^FF0000Tier A (Endgame / 4 Stones)^000000");

			.@page_items = 0;
			for (.@i = 0; .@i < $@WF_TotalDuns; .@i++) {
				if ($@WF_Dun_Tier[.@i] == .@view_tier) {
					if (.@page_items == 0) {
						mes "=== " + .@tier_title$ + " ===";
					}
					if (getd("#WAYFARER_DUNGEONS_" + (.@i / 30)) & (1 << (.@i % 30))) {
						mes "  ^00AA00[Unlocked]^000000 " + $@WF_Dun_Name$[.@i];
					} else {
						mes "  ^777777[Locked]^000000 " + $@WF_Dun_Name$[.@i] + " ^888888(" + $@WF_Dun_Cost[.@i] + " " + $@WF_Dun_City$[.@i] + ")^000000";
					}
					.@page_items++;
					if (.@page_items == 5) {
						.@page_items = 0;
						next;
					}
				}
			}
			if (.@page_items > 0) {
				next;
			}
		} else if (.@sel == 5) {
			for (.@t = 1; .@t <= 4; .@t++) {
				.@tier_title$ = ((.@t == 1) ? "^00AA00Tier D (Novice / 1 Stone)^000000" :
				                 (.@t == 2) ? "^0055FFTier C (Standard / 2 Stones)^000000" :
				                 (.@t == 3) ? "^FF8800Tier B (Advanced / 3 Stones)^000000" :
				                               "^FF0000Tier A (Endgame / 4 Stones)^000000");
				.@page_items = 0;
				for (.@i = 0; .@i < $@WF_TotalDuns; .@i++) {
					if ($@WF_Dun_Tier[.@i] == .@t) {
						if (.@page_items == 0) {
							mes "=== " + .@tier_title$ + " ===";
						}
						if (getd("#WAYFARER_DUNGEONS_" + (.@i / 30)) & (1 << (.@i % 30))) {
							mes "  ^00AA00[Unlocked]^000000 " + $@WF_Dun_Name$[.@i];
						} else {
							mes "  ^777777[Locked]^000000 " + $@WF_Dun_Name$[.@i] + " ^888888(" + $@WF_Dun_Cost[.@i] + " " + $@WF_Dun_City$[.@i] + ")^000000";
						}
						.@page_items++;
						if (.@page_items == 5) {
							.@page_items = 0;
							next;
						}
					}
				}
				if (.@page_items > 0) {
					next;
				}
			}
		}
	}
}

// =============================================================
// Field Wayfarer Warper Service - Regional Guild Navigators
// Allows travel to unlocked dungeon entrances from any major town.
// Teleportation cost: 1 to 4 Wayfarer's Stones (ID 50001) by Tier
// =============================================================
-	script	WayfarerWarperBase	-1,{
	// Migrate legacy bitmask if present
	if (#WAYFARER_DUNGEONS) {
		#WAYFARER_DUNGEONS_0 = #WAYFARER_DUNGEONS_0 | #WAYFARER_DUNGEONS;
		#WAYFARER_DUNGEONS = 0;
	}

	if (getarraysize($@WF_Dun_Name$) == 0)
		callsub "Wayfarer Master::OnInit";

	mes "[^0055FFWayfarer Warper^000000]";
	mes "Greetings, adventurer!";
	mes "I am a field teleportation officer of the ^0055FFWayfarer Guild^000000.";
	mes "I can transport you to the entrance of any dungeon unlocked on your account.";
	mes "Our regional teleportation matrix requires ^00AA00Wayfarer's Stones^000000 depending on the destination tier (1 to 4 stones).";
	next;

	while (1) {
		mes "[^0055FFWayfarer Warper^000000]";
		mes "How can I assist your travels today?";
		next;

		set .@menu, select(
			"Travel to Unlocked Dungeon",
			"Travel to Adventurer Agency (Izlude)",
			"Inspect Account Waypoints",
			"Cancel"
		);

		switch (.@menu) {
		case 1: // Travel to Unlocked Dungeon
			.@dun_menu$ = "";
			for (.@a = 0; .@a < $@WF_TotalDuns; .@a++) {
				.@i = $@WF_Alpha_Idx[.@a];
				if (getd("#WAYFARER_DUNGEONS_" + (.@i / 30)) & (1 << (.@i % 30))) {
					.@dun_menu$ = .@dun_menu$ + "^000000" + $@WF_Dun_Name$[.@i] + "^000000:";
				} else {
					.@dun_menu$ = .@dun_menu$ + "^FF0000" + $@WF_Dun_Name$[.@i] + "^000000:";
				}
			}
			.@dun_menu$ = .@dun_menu$ + "Back";

			mes "[^0055FFWayfarer Warper^000000]";
			mes "Select your destination entrance:";
			mes "(^000000Available in Black^000000 | ^FF0000Locked in Red^000000)";
			next;

			.@chosen = select(.@dun_menu$) - 1;
			if (.@chosen == $@WF_TotalDuns) continue;

			.@target_idx = $@WF_Alpha_Idx[.@chosen];
			.@req_stones = $@WF_Dun_Tier[.@target_idx];
			.@is_unlocked = (getd("#WAYFARER_DUNGEONS_" + (.@target_idx / 30)) & (1 << (.@target_idx % 30))) ? 1 : 0;

			if (!.@is_unlocked) {
				mes "=== ^FF0000Locked Waypoint^000000 ===";
				mes "Destination: ^FF0000" + $@WF_Dun_Name$[.@target_idx] + "^000000";
				mes "Status: ^FF0000Locked^000000";
				mes "Required Travel Fee: ^00AA00" + .@req_stones + "x Wayfarer's Stone" + ((.@req_stones > 1) ? "s" : "") + "^000000";
				mes " ";
				mes "This waypoint has not yet been unlocked on your account.";
				mes "Visit the ^0055FFWayfarer Master^000000 at the Adventurer Agency in Izlude to license new waypoints.";
				mes "Unlock Fee: ^00AA00" + $@WF_Dun_Cost[.@target_idx] + "x " + $@WF_Dun_City$[.@target_idx] + " Coins^000000.";
				next;
				continue;
			}

			mes "=== ^0055FFWaypoint Confirmation^000000 ===";
			mes "Destination: ^0055FF" + $@WF_Dun_Name$[.@target_idx] + "^000000";
			mes "Entrance: ^777777" + $@WF_Dun_Map$[.@target_idx] + "^000000";
			mes "Required Fee: ^00AA00" + .@req_stones + "x Wayfarer's Stone" + ((.@req_stones > 1) ? "s" : "") + "^000000";
			mes "You currently have: ^0055FF" + countitem(50001) + "x Wayfarer's Stone(s)^000000";
			mes " ";
			mes "Would you like to teleport now?";
			next;

			if (select("Depart now:Back") == 2) {
				continue;
			}

			if (countitem(50001) < .@req_stones) {
				mes "[^0055FFWayfarer Warper^000000]";
				mes "You do not have enough ^0055FFWayfarer's Stones^000000.";
				mes "You need ^FF0000" + .@req_stones + "^000000 stone" + ((.@req_stones > 1) ? "s" : "") + ", but only possess ^FF0000" + countitem(50001) + "^000000.";
				next;
				continue;
			}

			delitem 50001, .@req_stones;
			specialeffect2 EF_TELEPORT;
			specialeffect2 EF_INCAGILITY;
			mes "[^0055FFWayfarer Warper^000000]";
			mes "Opening portal to ^0055FF" + $@WF_Dun_Name$[.@target_idx] + "^000000 entrance!";
			mes "Safe travels, adventurer!";
			close2;
			warp $@WF_Dun_Map$[.@target_idx], $@WF_Dun_X[.@target_idx], $@WF_Dun_Y[.@target_idx];
			end;

		case 2: // Travel to Adventurer Agency
			if (strcharinfo(3) == "iz_ac01") {
				mes "[^0055FFWayfarer Warper^000000]";
				mes "You are already inside the Adventurer Agency Headquarters!";
				mes "Would you like to step outside to Izlude Harbor?";
				next;
				if (select("Exit to Izlude:Stay here") == 1) {
					mes "[^0055FFWayfarer Warper^000000]";
					mes "Safe journeys!";
					close2;
					specialeffect2 EF_TELEPORT;
					warp "izlude", 130, 70;
					end;
				}
				continue;
			}
			mes "[^0055FFWayfarer Warper^000000]";
			mes "Would you like to visit the ^0055FFAdventurer Agency Headquarters^000000 in Izlude?";
			next;
			if (select("Depart for Agency:Stay here") == 1) {
				mes "[^0055FFWayfarer Warper^000000]";
				mes "Transporting you to the Agency Headquarters...";
				close2;
				specialeffect2 EF_TELEPORT;
				warp "iz_ac01", 103, 40;
				end;
			}
			continue;

		case 3: // Inspect
			callfunc "F_WF_InspectDungeons";
			continue;

		case 4: // Cancel
			mes "[^0055FFWayfarer Warper^000000]";
			mes "May the winds favor your hunts.";
			close;
		}
	}
}
