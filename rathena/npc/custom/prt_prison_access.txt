// --------------------------------------------------------------
// Knight-Lieutenant Raines (prt_castle 123, 65) - Sprite 470
// --------------------------------------------------------------
prt_castle,123,65,2	script	Raines#prt_prison	470,{
	mes "[Knight-Lieutenant Raines]";
	mes "Salute, adventurer!";
	mes "Centuries ago, these deep dungeons were sealed to contain an ancient, dark corruption.";
	mes "Now that the seals have ruptured and the darkness reawakens, our vanguard fought bravely to hold the line.";
	next;
	mes "[Knight-Lieutenant Raines]";
	mes "By recovering the seals of our vanguard Knights who fell in this new crisis, you honor their immediate sacrifice.";
	mes "The Royal Guard grants you access to our specialized armory in exchange for ^0000FFKnight's Seals^000000.";
	mes " ";
	mes "You currently hold: ^00AA00" + countitem(50024) + " Knight's Seals^000000.";
	next;
	callshop "raines_prison_shop", 1;
	npcshopattach "raines_prison_shop";
	end;

OnBuyItem:
	.@size = getarraysize(@bought_nameid);
	.@total_cost = 0;

	for (.@i = 0; .@i < .@size; .@i++) {
		.@item_idx = inarray(.Shop_Data, @bought_nameid[.@i]);
		if (.@item_idx == -1) {
			dispbottom "System Error: Item not found in the supply line.";
			end;
		}
		.@total_cost += (.Shop_Data[.@item_idx + 1] * @bought_quantity[.@i]);
	}

	if (.@total_cost > countitem(50024)) {
		mes "[Knight-Lieutenant Raines]";
		mes "I am afraid you lack the necessary Knight's Seals for these supplies, my friend.";
		mes "Total cost: ^FF0000" + .@total_cost + "^000000 Seals.";
		close;
	}

	delitem 50024, .@total_cost;

	for (.@i = 0; .@i < .@size; .@i++) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
	}

	mes "[Knight-Lieutenant Raines]";
	mes "Thank you for your service. May these relics serve you well in the depths below.";
	specialeffect2 EF_MONEY;
	close;

OnInit:
	setarray .Shop_Data[0],
		2875, 100,
		2886, 100,
		2799, 100;

	npcshopdelitem "raines_prison_shop", 502;
	for (.@i = 0; .@i < getarraysize(.Shop_Data); .@i += 2) {
		npcshopadditem "raines_prison_shop", .Shop_Data[.@i], .Shop_Data[.@i+1];
	}
	end;
}

-	shop	raines_prison_shop	-1,502:-1
