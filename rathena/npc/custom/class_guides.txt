-	script	NvSwordman	-1,{
	mes "[Swordman Guildsman]";
	mes "Can I help you?";
	next;
	switch(select("About Swordman Class.:Job change to Swordman.:Got any quests?:Cancel.")) {
	case 1:
		mes "[Swordman Guildsman]";
		mes "Literally, Swordman means one, who is specialized in wielding swords.";
		mes "But they can also choose to use spears if they so choose.";
		next;
		mes "[Swordman Guildsman]";
		mes "We possesses strong physical strength. So naturally we can equip heavy armors and weapons.";
		next;
		if (isequipped(2819) || countitem(2819)) {
			mes "[Swordman Guildsman]";
			mes "Try the Swordman skills listed in the manual.";
			mes "Open your skill menu by pressing Alt + S";
			close;
		}
		mes "[Swordman Guildsman]";
		mes "If you're interested in being a Swordman, I'll offer you this manual.";
		mes "Do you want to try the skills of a Swordman?";
		next;
		if (select("I'll try the Swordman skills.:No thanks.") == 1) {
			mes "[Swordman Guildsman]";
			mes "Here is the Swordman manual.";
			next;
			callfunc "F_NvErase";
			getitem 2819,1; //Swordman_Manual
			close;
		}
		close;
	case 2:
		mes "[Swordman Guildsman]";
		mes "Do you really want to change your job to Swordman?";
		next;
		if (select("I'll consider it again.:I'll go to the Swordman Association.") == 2) {
			mes "[Swordman Guildsman]";
			mes "Then I'll end your training process and send you to the Swordman Guild.";
			callfunc "F_NvErase",1;
			close2;
			savepoint "izlude",128,98,1,1;
			warp "izlude",52,136;
			end;
		}
		close;
	case 3:
		set .@quest, checkquest(7123);
		if (.@quest == -1) {
			mes "[Swordman Guildsman]";
			mes "Hunt a couple of Picky monsters around here?";
			setquest 7123;
			close;
		} else if (checkquest(7123,HUNTING) == 2) {
			mes "[Swordman Guildsman]";
			mes "Great. I'll give you some potions as a reward.";
			completequest 7123;
			getitem 569,200; //Novice_Potion
			getexp 100,100;
			close;
		} else {
			mes "[Swordman Guildsman]";
			mes "Hunt two Picky monsters.";
			close;
		}
	case 4:
		close;
	}
}

			close;
		}
		
		set .@hunt, checkquest(7124,HUNTING);
		set .@items, countitem(705);
		
		if (.@hunt == 2 && .@items >= 5) {
			mes "[Mage Guild Member]";
			mes "Excellent job! You've proven your worth both in combat and in gathering reagents.";
			next;
			delitem 705,5; // Clover
			completequest 7124;
			getexp 100,100;
			getitem 12324,30; // N_Butterfly_Wing
			mes "[Mage Guild Member]";
			mes "Here is your reward. Good luck with your magical studies!";
			close;
		}
		
		mes "[Mage Guild Member]";
		mes "You haven't finished yet!";
		if (.@hunt != 2) mes "- You still need to hunt all ^FF00005 Lunatics^000000 (State: "+.@hunt+").";
		if (.@items < 5) mes "- You still need to bring ^0000FF"+(5-.@items)+" more Clovers^000000.";
		close;
	case 4:
		mes "[Mage Guild Member]";
		mes "Resetting your progress for the Mage trial...";
		erasequest 7124;
		mes "Done. You can take the quest again now.";
		close;
	case 5:
		close;
	}
}

// Support Function
function	script	F_NvErase	{
	if (getarg(0,0)) {
		if (checkquest(7122) > -1) erasequest 7122;
		if (checkquest(7123) > -1) erasequest 7123;
		if (checkquest(7124) > -1) erasequest 7124;
		if (checkquest(7126) > -1) erasequest 7126;
		if (checkquest(7127) > -1) erasequest 7127;
	}
	nude;
	setarray .@manuals[0],countitem(2819),countitem(2820),countitem(2821),countitem(2822),countitem(2823),countitem(2824);
	if (.@manuals[0]) delitem 2819,.@manuals[0];
	if (.@manuals[1]) delitem 2820,.@manuals[1];
	if (.@manuals[2]) delitem 2821,.@manuals[2];
	if (.@manuals[3]) delitem 2822,.@manuals[3];
	if (.@manuals[4]) delitem 2823,.@manuals[4];
	if (.@manuals[5]) delitem 2824,.@manuals[5];
	return;
}
