	mes "[Reset Specialist]";
	mes "Welcome, adventurer!";
	mes "During our ^0055FFClosed Beta^000000 phase, all stat and skill resets are ^00AA00completely free^000000.";
	mes " ";
	mes "What would you like to reset today?";
	next;

	switch(select("Reset Skill Points:Reset Stat Points:Reset Both (Stats & Skills):Cancel")) {
		case 1:
			mes "[Reset Specialist]";
			mes "Are you sure you want to reset all your ^0055FFSkill Points^000000?";
			next;
			if (select("Yes, reset my skills!:No, let me think") == 2) close;
			
			sc_end_class;
			resetskill;

			mes "Your skill points have been fully reset! Allocate them wisely.";
			close;

		case 2:
			mes "[Reset Specialist]";
			mes "Are you sure you want to reset all your ^0055FFStat Points^000000?";
			next;
			if (select("Yes, reset my stats!:No, let me think") == 2) close;
			
			resetstatus;

			mes "Your stat points have been fully reset!";
			close;

		case 3:
			mes "[Reset Specialist]";
			mes "Are you sure you want to reset ^0055FFBOTH Stats and Skills^000000?";
			next;
			if (select("Yes, reset both!:No, let me think") == 2) close;
			
			sc_end_class;
			resetskill;
			resetstatus;
			
			mes "[Reset Specialist]";
			mes "Both your stats and skills have been completely reset!";
			close;

		case 4:
			mes "[Reset Specialist]";
			mes "Return whenever you need to adjust your build!";
			close;
	}
}
