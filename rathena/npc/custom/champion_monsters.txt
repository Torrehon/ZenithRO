OnNPCKillEvent:
	// 1. ATTACH PLAYER
	attachrid(killerrid);

	set .@id, killedrid;
	set .@map$, strcharinfo(3);

	// 2. RESTRICCIÓN DE MAPA: Si el mapa tiene un monstruo asignado, sólo las bajas de ese monstruo suman contador.
	set .@allowed_mob, getd(".map_mob_" + .@map$);
	if (.@allowed_mob <= 0 || .@id != .@allowed_mob) end;

	// 3. RECOLECTAR CHAMPION IDS DEFINIDOS PARA ESTE MOB
	deletearray .@ch_list[0];
	for (set .@i, 1; .@i <= 5; set .@i, .@i + 1) {
		set .@c, getd(".ch_" + .@id + "_" + .@i);
		if (.@c > 0)
			setarray .@ch_list[getarraysize(.@ch_list)], .@c;
	}

	// Filtro: Solo mobs con champions mapeados
	if (getarraysize(.@ch_list) == 0) end;

	// 4. REQUISITOS DINÁMICOS POR NIVEL
	set .@lvl, getmonsterinfo(.@id, MOB_LV);
	if (.@lvl <= 0) set .@lvl, getmonsterinfo(killedrid, MOB_LV);

	// LÓGICA DE NIVELES
	if (.@lvl <= 25) { set .@req, 200; }
	else if (.@lvl <= 50) { set .@req, 150; }
	else if (.@lvl <= 75) { set .@req, 100; }
	else { set .@req, 50; }

	// 5. LÓGICA DE CONTADOR
	set .@var$, "$CK_" + .@map$ + "_" + .@id;
	setd .@var$, getd(.@var$) + 1;
	set .@curr, getd(.@var$);
	
	// 6. MOSTRAR PROGRESO POR HITOS (Cada 50 kills o en la recta final <= 25)
	set .@rem, .@req - .@curr;
	if ((.@curr % 50 == 0) || (0 <= .@rem && .@rem <= 25)) {
		dispbottom "Champion Progress [" + getmonsterinfo(.@id, MOB_NAME) + "]: " + .@curr + "/" + .@req;
	}

	// 7. SPAWN LOGIC (Random Map Spawn + Minimap Indicator)
	if (.@curr >= .@req) {
		setd .@var$, 0; 

		// Selecciona un Champion válido de la lista de forma totalmente aleatoria
		set .@champ, .@ch_list[rand(getarraysize(.@ch_list))];

		// Invoca al monstruo en una casilla aleatoria del mapa
		monster .@map$, 0, 0, "--en--", .@champ, 1, "CHAMPION_SYSTEM::OnDeath";

		if ($@mobid[0] > 0) {
			// Obtiene la posición X e Y del Champion recién invocado
			getunitdata $@mobid[0], .@u;
			set .@cx, .@u[UMOB_X];
			set .@cy, .@u[UMOB_Y];

			// Marca la ubicación en el minimapa de todos los jugadores presentes en el mapa
			if (.@cx > 0 && .@cy > 0)
				viewpointmap .@map$, 1, .@cx, .@cy, 1, 0xFF0000;

			mapannounce .@map$, "A Champion (" + getmonsterinfo(.@champ, MOB_NAME) + ") has appeared in a random location on the map!", bc_map, 0xFFFF00;
		}
	}
	end;

OnDeath:
	// Borra la marca del minimapa al ser derrotado
	getmapxy(.@m$, .@x, .@y);
	viewpointmap .@m$, 2, .@x, .@y, 1, 0xFF0000;

	// Dynamic Drop (20-50%)
	set .@lvl, getmonsterinfo(killedrid, MOB_LV);
	set .@drop, 20 + (.@lvl * 30 / 100);
	if (.@drop > 50) set .@drop, 50;

	if (rand(100) < .@drop) {
		getitem 12106, 1;
		dispbottom "Rare item obtained! (" + .@drop + "% chance).";
	}
	
	announce "The Champion (" + getmonsterinfo(killedrid, MOB_NAME) + ") was defeated by " + strcharinfo(0) + "!", bc_map, 0x00FF00;
	end;
