		if (checkquest(80040) == -1) { setquest 80040; }
		mes "Thank you! Please defeat ^FF000010 Marcs^000000, ^FF000010 Swordfish^000000, and ^FF000010 Phens^000000 on the upper floors.";
		next;
		mes "[Archeologist Dan]";
		mes "Once done, you don't need to return to the surface. Head straight down to the 3rd floor and report directly to the Chief Archeologist.";

		close;
	}

	// Player has Trident of the Depths (50010)
	if (getcharid(1) == 0) {
		mes "You have found the Trident! Excellent!";
		mes "However, we need a party to breach the seal and secure the area.";
		close;
	}

	// ==========================================================
	// 1. EL "FIX": GATEWAY ABIERTO Y LA INSTANCIA YA CREADA
	// Permite reingresar a cualquier miembro
	// ==========================================================
	if (instance_id(IM_PARTY) > 0) {
		mes "^888888*A glowing purple gateway is open here. You can hear faint wails from inside.*^000000";
		next;
		if (select("Enter the gateway.:Wait.") == 1) {
			close2;
			instance_enter("Byalan");
			end;
		}
		close;
	}

	if (!is_party_leader()) {
		mes "Only your party leader can initiate the breach of the seal.";
		close;
	}

	mes "You have the Trident of the Depths! Are you ready to open the gateway and face the guardian?";
	next;
	if (select("Open Gateway.:Wait.") == 1) {
		if (instance_id(IM_PARTY) <= 0) {
			if (instance_create("Byalan") < 0) {
				mes "Gateway failure. (Instance error).";
				close;
			}
		}
		close2;
		instance_enter("Byalan");
		end;
	}
	close;
}

// --- 3. INSIDE INSTANCE (1@iz_dun03) ---
1@iz_dun03,30,170,4	script	Chief Archeologist#iz	804,{
	if ('boss_dead == 1) {
		mes "[Chief Archeologist]";
		mes "Thank you for saving us! The Mermaid of the Depths has been defeated.";
		next;
		if (byalan_quest == 2 || byalan_quest == 3) {
			mes "[Chief Archeologist]";
			mes "We must return to the surface immediately. Please report to Dan at the dungeon entrance in the center of the island to claim your rewards and conclude the expedition.";
			next;
			
			// Sincronización de quest
			.@origin_aid = getcharid(3);
			getpartymember getcharid(1), 2;
			for (.@i = 0; .@i < $@partymembercount; .@i++) {
				if (attachrid($@partymemberaid[.@i])) {
					if (byalan_quest < 3) { set byalan_quest, 3; }
					if (checkquest(80041) == 0 || checkquest(80041) == 1) { completequest 80041; }
					if (checkquest(80042) == 0 || checkquest(80042) == 1) { completequest 80042; }
					if (checkquest(80043) == -1) { setquest 80043; }
					detachrid;
				}
			}
			attachrid(.@origin_aid);
		}
		
		mes "[Chief Archeologist]";
		mes "Would you like to return to the surface?";
		next;
		if (select("Yes, please.:Stay here.") == 1) {
			warp "izlu2dun", 114, 88;
		}
		close;
	}

	mes "[Chief Archeologist]";
	mes "Defeat the sea creatures to secure the area!";
	close;

OnInstanceInit:
	setmapflag instance_mapname("1@iz_dun03"), mf_noteleport;
	setmapflag instance_mapname("1@iz_dun03"), mf_nomemo;
	'boss_dead = 0;
	'mobs_alive = 0;
	'wave = 0;
	initnpctimer;
	end;

OnTimer1000:
	mapannounce instance_mapname("1@iz_dun03"), "Archeologist Chief: The seal is broken! Protect the camp from the upcoming horde!", bc_map, "0xFFFF00";
	end;

OnTimer4000:
	mapannounce instance_mapname("1@iz_dun03"), "A horde of aggressive sea creatures is approaching!", bc_map, "0xFF0000";
	// Spawn Wave 1: 5 Vadon (agresivos), 2 Obeaune, 1 Marse (agresivo)
	'wave = 1;
	'mobs_alive = 8;
	
	setarray .@spawn_x[0], 32, 34, 37, 40, 42, 38, 34, 33;
	setarray .@spawn_y[0], 177, 179, 177, 173, 168, 162, 161, 163;
	
	// Vadon (ID 1066) - Spawn 5
	for (.@i = 0; .@i < 5; .@i++) {
		monster instance_mapname("1@iz_dun03"), .@spawn_x[.@i], .@spawn_y[.@i], "Aggressive Vadon", 1066, 1, instance_npcname("Chief Archeologist#iz")+"::OnMobKilled";
		setunitdata $@mobid[0], UMOB_MODE, (MD_CANMOVE | MD_AGGRESSIVE | MD_CANATTACK);
	}
	
	// Obeaune (ID 1044) - Spawn 2
	for (.@i = 5; .@i < 7; .@i++) {
		monster instance_mapname("1@iz_dun03"), .@spawn_x[.@i], .@spawn_y[.@i], "Aggressive Obeaune", 1044, 1, instance_npcname("Chief Archeologist#iz")+"::OnMobKilled";
		setunitdata $@mobid[0], UMOB_MODE, (MD_CANMOVE | MD_AGGRESSIVE | MD_CANATTACK);
	}
	
	// Marse (ID 1144) - Spawn 1
	monster instance_mapname("1@iz_dun03"), .@spawn_x[7], .@spawn_y[7], "Aggressive Marse", 1144, 1, instance_npcname("Chief Archeologist#iz")+"::OnMobKilled";
	setunitdata $@mobid[0], UMOB_MODE, (MD_CANMOVE | MD_AGGRESSIVE | MD_CANATTACK);
	
	stopnpctimer;
	end;

OnMobKilled:
	'mobs_alive--;
	if ('mobs_alive <= 0) {
		if ('wave == 1) {
			'wave = 2;
			'mobs_alive = 6;
			mapannounce instance_mapname("1@iz_dun03"), "The sea grows more violent! A second wave is emerging!", bc_map, "0xFF0000";
			
			setarray .@spawn_x[0], 32, 34, 37, 40, 42, 38, 34, 33;
			setarray .@spawn_y[0], 177, 179, 177, 173, 168, 162, 161, 163;
			
			// Wave 2: 2 Marc (ID 1045), 4 Swordfish (ID 1069)
			for (.@i = 0; .@i < 2; .@i++) {
				monster instance_mapname("1@iz_dun03"), .@spawn_x[.@i], .@spawn_y[.@i], "Aggressive Marc", 1045, 1, instance_npcname("Chief Archeologist#iz")+"::OnMobKilled";
			}
			for (.@i = 2; .@i < 6; .@i++) {
				monster instance_mapname("1@iz_dun03"), .@spawn_x[.@i], .@spawn_y[.@i], "Aggressive Swordfish", 1069, 1, instance_npcname("Chief Archeologist#iz")+"::OnMobKilled";
			}
			
		} else if ('wave == 2) {
			'wave = 3;
			mapannounce instance_mapname("1@iz_dun03"), "The waters grow calm... but a colossal presence approaches...", bc_map, "0xFFFF00";
			sleep 4000;
			mapannounce instance_mapname("1@iz_dun03"), "The Mermaid of the Depths emerges from the abyss!", bc_map, "0xFF0000";
			soundeffectall "earth_quake.wav", 0, instance_mapname("1@iz_dun03");
			
			monster instance_mapname("1@iz_dun03"), 37, 177, "Mermaid of the Depths", 20647, 1, instance_npcname("Chief Archeologist#iz")+"::OnBossKilled";
			.@mob_gid = $@mobid[0];
			if (.@mob_gid > 0) { setunitdata .@mob_gid, UMOB_SIZE, 8; }
		}
	}
	end;

OnBossKilled:
	'boss_dead = 1;
	mapannounce instance_mapname("1@iz_dun03"), "The Mermaid of the Depths has been defeated! The area is safe.", bc_map, "0x00FF00";
	
	// Sincronizar quest
	if (getcharid(1) > 0) .party_id = getcharid(1);
	.@origin_aid = getcharid(3);
	getpartymember .party_id, 2;
	for (.@i = 0; .@i < $@partymembercount; .@i++) {
		if (attachrid($@partymemberaid[.@i])) {
			if (byalan_quest == 2) { set byalan_quest, 3; }
			if (checkquest(80042) == 0 || checkquest(80042) == 1) { completequest 80042; }
			detachrid;
		}
	}
	attachrid(.@origin_aid);
	end;
}
