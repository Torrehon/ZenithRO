// --------------------------------------------------------------
// Helper Function: Checks if player has a Dawn Weapon equipped
// --------------------------------------------------------------
function	script	F_HasDawnWeaponEquipped	{
	.@r = getequipid(EQI_HAND_R);
	.@l = getequipid(EQI_HAND_L);
	if ((.@r >= 40000 && .@r <= 40011) || (.@l >= 40000 && .@l <= 40011))
		return 1;
	return 0;
}

// --------------------------------------------------------------
// Helper Function: Checks if player is 2nd Job (Normal or Trans)
// --------------------------------------------------------------
function	script	F_Is2ndJob	{
	switch (BaseJob) {
		case Job_Knight:
		case Job_Priest:
		case Job_Wizard:
		case Job_Blacksmith:
		case Job_Hunter:
		case Job_Assassin:
		case Job_Crusader:
		case Job_Monk:
		case Job_Sage:
		case Job_Rogue:
		case Job_Alchemist:
		case Job_Bard:
		case Job_Dancer:
			return 1;
		default:
			return 0;
	}
}

// --------------------------------------------------------------
// Helper Function: Checks if player already has a Soul skill learned
// --------------------------------------------------------------
function	script	F_HasSoulSkill	{
	if (getskilllv(getarg(0)) > 0 || getskilllv(getarg(1)) > 0)
		return 1;
	return 0;
}

// Helper Function for Energy Nodes
function	script	F_ChannelEnergyNode	{
	.@spot_mask = getarg(0);
	.@spot_num  = getarg(1);
	.@location$ = getarg(2);

	if (SOUL_QUEST1_STEP < 1) {
		mes "[Mana Sanctuary]";
		mes "An ancient mana nexus. It radiates a faint primordial energy, but you currently have no reason to channel it.";
		close;
	}

	if (!F_HasDawnWeaponEquipped()) {
		mes "[Mana Sanctuary]";
		mes "You feel the magical energy surging from the earth, but you do not have a ^0000FFDawn Weapon^000000 equipped to absorb it.";
		close;
	}

	if (SOUL_QUEST1_SPOTS & .@spot_mask) {
		mes "[Mana Sanctuary]";
		mes "You have already absorbed the magical energy from this sanctuary into your Dawn Weapon.";
		close;
	}

	mes "[Mana Sanctuary]";
	mes "You touch the ancient mana sanctuary and begin channeling its " + .@location$ + " magic energy into your Dawn Weapon...";
	close2;

	progressbar "0x00FF00", 3;

	SOUL_QUEST1_SPOTS |= .@spot_mask;
	specialeffect2 EF_HOLYLIGHT;

	// Calculate count of charged spots
	.@count = 0;
	if (SOUL_QUEST1_SPOTS & 1) .@count++;
	if (SOUL_QUEST1_SPOTS & 2) .@count++;
	if (SOUL_QUEST1_SPOTS & 4) .@count++;
	if (SOUL_QUEST1_SPOTS & 8) .@count++;

	dispbottom "Your Dawn weapon absorbs the " + .@location$ + " magic energy! (" + .@count + "/4)";

	if (SOUL_QUEST1_SPOTS == 15) {
		dispbottom "Your Dawn weapon is fully charged with magic energy! Return to the Scholar in Prontera library.";
		if (checkquest(90001) == 1) {
			completequest 90001;
			setquest 90002;
		}
	}
	end;
}
