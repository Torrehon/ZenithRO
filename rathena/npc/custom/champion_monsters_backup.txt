OnNPCKillEvent:
	// 1. ATTACH PLAYER
	attachrid(killerrid);

	set .@id, killedrid;
	set .@map$, strcharinfo(3);

	// Filter: Only mobs with champions
	if (getd(".ch_" + .@id + "_1") == 0) end;

	// 2. DYNAMIC REQUIREMENTS BY LEVEL
	// We use MOB_LV constant for better compatibility
	set .@lvl, getmonsterinfo(.@id, MOB_LV);
	
	// If it still returns 0, we try to get it via ID directly
	if (.@lvl <= 0) set .@lvl, getmonsterinfo(killedrid, MOB_LV);

	// --- DEBUG LINE (Uncomment if you want to see the detected level in chat) ---
	// dispbottom "DEBUG: Mob Level detected: " + .@lvl;
	// ----------------------------------------------------------------------------

	// LEVEL LOGIC
	if (.@lvl <= 25) { set .@req, 300; }
	else if (.@lvl <= 50) { set .@req, 200; }
	else if (.@lvl <= 75) { set .@req, 150; }
	else { set .@req, 75; }

	// 3. COUNTER LOGIC
	set .@var$, "$CK_" + .@map$ + "_" + .@id;
	setd .@var$, getd(.@var$) + 1;
	set .@curr, getd(.@var$);
	
	// 4. PROGRESS DISPLAY (145/150 Format)
	set .@rem, .@req - .@curr;
	if (.@rem >= 0 && .@rem <= 25) {
		dispbottom "Champion Progress [ID: " + .@id + "]: " + .@curr + "/" + .@req;
	}

	// 5. SPAWN LOGIC
	if (.@curr >= .@req) {
		setd .@var$, 0; 

		set .@v, 0;
		for (set .@i, 1; .@i <= 5; set .@i, .@i + 1) {
			if (getd(".ch_" + .@id + "_" + .@i) > 0) set .@v, .@v + 1;
		}
		set .@champ, getd(".ch_" + .@id + "_" + (rand(.@v) + 1));

		getmapxy(.@map$, .@px, .@py);
		if (.@px <= 0) { set .@px, 150; set .@py, 150; }

		set .@x1, .@px - 10; set .@y1, .@py - 10;
		set .@x2, .@px + 10; set .@y2, .@py + 10;

		areamonster .@map$, .@x1, .@y1, .@x2, .@y2, "--en--", .@champ, 1, "CHAMPION_SYSTEM::OnDeath";
		
		mapannounce .@map$, "A Champion has appeared near " + strcharinfo(0) + "!", bc_map, 0xFFFF00;
	}
	end;

OnDeath:
	// Dynamic Drop (20-50%)
	set .@lvl, getmonsterinfo(killedrid, MOB_LV);
	set .@drop, 20 + (.@lvl * 30 / 100);
	if (.@drop > 50) set .@drop, 50;

	if (rand(100) < .@drop) {
		getitem 12106, 1;
		dispbottom "Rare item obtained! (" + .@drop + "% chance).";
	}
	
	announce "The Champion was defeated by " + strcharinfo(0) + "!", bc_map, 0x00FF00;
	end;
