// --------------------------------------------------------------
// Helper Function: F_GetEquippedDawnWeapon
// Returns the Item ID of any Dawn Weapon equipped by the player (40000-40011)
// --------------------------------------------------------------
function	script	F_GetEquippedDawnWeapon	{
	.@wep_id = getequipid(EQI_HAND_R);
	if (.@wep_id >= 40000 && .@wep_id <= 40011) return .@wep_id;
	.@wep_id = getequipid(EQI_HAND_L);
	if (.@wep_id >= 40000 && .@wep_id <= 40011) return .@wep_id;
	return 0;
}

// --------------------------------------------------------------
// Helper Function: F_CraftTwilightWeapon
// Args:
//   getarg(0): Twilight Weapon ID
//   getarg(1): Gem 1 ID (0 if none)
//   getarg(2): Gem 1 Amount
//   getarg(3): Gem 2 ID (0 if none)
//   getarg(4): Gem 2 Amount
//   getarg(5): Gem 3 ID (0 if none)
//   getarg(6): Gem 3 Amount
// --------------------------------------------------------------
function	script	F_CraftTwilightWeapon	{
	.@twilight_id = getarg(0);
	.@gem1        = getarg(1, 0);
	.@count1      = getarg(2, 0);
	.@gem2        = getarg(3, 0);
	.@count2      = getarg(4, 0);
	.@gem3        = getarg(5, 0);
	.@count3      = getarg(6, 0);

	// 1. Detect any equipped Dawn Weapon (Living Star Fragment)
	.@dawn_id = callfunc("F_GetEquippedDawnWeapon");
	if (.@dawn_id == 0) {
		mes "[Einherjar]";
		mes "Please equip your ^0000FFDawn Weapon^000000!";
		mes "I must analyze the equipped Star Fragment to safely extract any cards and handle its Random Options.";
		close;
	}

	.@eq_slot = EQI_HAND_R;
	if (getequipid(EQI_HAND_R) != .@dawn_id) {
		if (getequipid(EQI_HAND_L) == .@dawn_id) {
			.@eq_slot = EQI_HAND_L;
		}
	}

	// 2. Weight & Space Check (Anti-Drop/Loss)
	if (!checkweight(.@twilight_id, 1)) {
		mes "[Einherjar]";
		mes "Your inventory is too heavy or full to receive this weapon!";
		mes "Free up inventory space and weight before proceeding.";
		close;
	}

	// 3. Weapon Transformation Confirmation
	mes "[Einherjar]";
	mes "Are you certain you wish to awaken your equipped " + mesitemlink(.@dawn_id) + " into " + mesitemlink(.@twilight_id) + "?";
	mes "^FF0000This will permanently transform the shape of your Star Fragment!^000000";
	next;
	if (select("Yes, forge " + getitemname(.@twilight_id) + "!:No, let me consider again") == 2) {
		mes "[Einherjar]";
		mes "Take your time, hero. Speak to me again when you are ready.";
		close;
	}

	// 4. Display Requirements & Option Choice
	mes "[Einherjar]";
	mes "To evolve your equipped " + mesitemlink(.@dawn_id) + " into " + mesitemlink(.@twilight_id) + ", I require:";
	mes "- 1x " + mesitemlink(.@dawn_id) + " (Equipped)";
	mes "- 10x " + mesitemlink(984);
	mes "- 20x " + mesitemlink(999);
	if (.@gem1 > 0) mes "- " + .@count1 + "x " + mesitemlink(.@gem1);
	if (.@gem2 > 0) mes "- " + .@count2 + "x " + mesitemlink(.@gem2);
	if (.@gem3 > 0) mes "- " + .@count3 + "x " + mesitemlink(.@gem3);
	mes "^888888(Tip: Click any item link above to inspect its details!)^000000";
	next;

	mes "[Einherjar]";
	mes "How would you like to handle the Star Fragment's Random Options?";
	mes "- ^0000FFKeep Current Options^000000: Transfer exact Random Options from your Dawn Weapon to the new Twilight Weapon.";
	mes "- ^0000FFRe-roll New Options^000000: Generate completely fresh Random Options for the Twilight Weapon.";
	next;
	.@opt_choice = select("Keep Current Random Options:Re-roll New Random Options:Cancel");
	if (.@opt_choice == 3) {
		mes "[Einherjar]";
		mes "Return when you are ready to unleash the Star Fragment's true potential.";
		close;
	}

	// 4. ATOMIC RE-VERIFICATION (Anti-Trade / Anti-Storage Exploit)
	if (getequipid(.@eq_slot) != .@dawn_id || countitem(984) < 10 || countitem(999) < 20) {
		mes "[Einherjar]";
		mes "Transaction aborted! You do not possess all required base materials in your inventory.";
		close;
	}
	if (.@gem1 > 0 && (countitem(.@gem1) - isequippedcnt(.@gem1)) < .@count1) { mes "[Einherjar]"; mes "Transaction aborted! You do not possess the required unequipped birthstones."; close; }
	if (.@gem2 > 0 && (countitem(.@gem2) - isequippedcnt(.@gem2)) < .@count2) { mes "[Einherjar]"; mes "Transaction aborted! You do not possess the required unequipped birthstones."; close; }
	if (.@gem3 > 0 && (countitem(.@gem3) - isequippedcnt(.@gem3)) < .@count3) { mes "[Einherjar]"; mes "Transaction aborted! You do not possess the required unequipped birthstones."; close; }


	// 5. EXTRACT & RETURN ALL EQUIPPED CARDS (Safe extraction)
	for (.@c = 0; .@c < 4; ++.@c) {
		.@card_id = getequipcardid(.@eq_slot, .@c);
		if (.@card_id > 0 && .@card_id < 40000) {
			getitem .@card_id, 1;
			mes "^0000FF[Card Extracted]^000000 " + mesitemlink(.@card_id) + " returned to your inventory.";
		}
	}

	// 6. READ RANDOM OPTIONS & DURABILITY
	.@current_durability = 100;
	for (.@o = 0; .@o < 5; ++.@o) {
		if (getequiprandomoption(.@eq_slot, .@o, ROA_ID) == 243) {
			.@current_durability = getequiprandomoption(.@eq_slot, .@o, ROA_VALUE);
		}
		if (.@opt_choice == 1) {
			.@opt_id[.@o]    = getequiprandomoption(.@eq_slot, .@o, ROA_ID);
			.@opt_val[.@o]   = getequiprandomoption(.@eq_slot, .@o, ROA_VALUE);
			.@opt_param[.@o] = getequiprandomoption(.@eq_slot, .@o, ROA_PARAM);
		}
	}

	// 7. ATOMIC DELETION AND ITEM DELIVERY
	delitem .@dawn_id, 1;
	delitem 984, 10;
	delitem 999, 20;
	if (.@gem1 > 0) delitem .@gem1, .@count1;
	if (.@gem2 > 0) delitem .@gem2, .@count2;
	if (.@gem3 > 0) delitem .@gem3, .@count3;

	getitem .@twilight_id, 1;
	SOUL_QUEST1_STEP = 5;

	// 8. APPLY OR REROLL RANDOM OPTIONS BASED ON PLAYER CHOICE
	getinventorylist;
	.@new_inv_idx = -1;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		if (@inventorylist_id[.@i] == .@twilight_id && @inventorylist_equip[.@i] == 0) {
			.@new_inv_idx = @inventorylist_idx[.@i];
			break;
		}
	}

	if (.@new_inv_idx >= 0) {
		// Reset all 5 option slots on newly crafted item first to prevent duplicate options from getitem
		for (.@o = 0; .@o < 5; ++.@o) {
			setitemoption .@new_inv_idx, .@o, 0, 0;
		}

		if (.@opt_choice == 1) { // Keep old options
			for (.@o = 0; .@o < 5; ++.@o) {
				if (.@opt_id[.@o] > 0) {
					setitemoption .@new_inv_idx, .@o, .@opt_id[.@o], .@opt_val[.@o];
				}
			}
		} else { // Reroll new options
			callfunc("F_ApplyTwilightRandomOptions", .@twilight_id, .@new_inv_idx, .@current_durability);
		}
	}
	equip .@twilight_id;

	specialeffect2 EF_REPAIRWEAPON;

	mes "[Einherjar]";
	mes "The Star Fragment blazes with celestial radiance!";
	mes "Receive your new weapon: " + mesitemlink(.@twilight_id) + "!";
	next;
	if (PRT_PRISON_QUEST == 0) {
		mes "[Einherjar]";
		mes "Champion, we the Einherjar have turned our gaze once more toward the mortal realm of Midgard...";
		mes "The Dawn Weapons were not the only star fragments that descended from the firmament centuries ago.";
		next;
		mes "[Einherjar]";
		mes "Other sentient star fragments fell across the world, but they harbored dark and malevolent intentions.";
		mes "We have sensed the disturbing influence of one such corrupt relic stirring within the capital itself, Prontera.";
		next;
		mes "[Einherjar]";
		mes "You must journey to the capital and investigate before the shadow spreads throughout the realm.";
		mes "Speak with the Grand Library Scholar in Prontera. He will know if strange rumors have surfaced recently.";
		PRT_PRISON_QUEST = 1;
	}
	close;
}

// --------------------------------------------------------------
// --------------------------------------------------------------
// Helper Function: F_ApplyTwilightRandomOptions
// Replicates server's exact C++ pc_apply_random_option engine (pc.cpp) for Twilight Weapons
// Applies 1-3 custom random options + Durability (ID 243, value .@durability)
// --------------------------------------------------------------
function	script	F_ApplyTwilightRandomOptions	{
	.@wep_id     = getarg(0);
	.@inv_idx    = getarg(1);
	.@durability = getarg(2, 100);

	if (.@inv_idx < 0) return;

	// Reset all 5 option slots first
	for (.@o = 0; .@o < 5; ++.@o) {
		setitemoption .@inv_idx, .@o, 0, 0;
	}

	// 1. Resolve Group ID matching pc_resolve_random_group in C++ (pc.cpp)
	.@group_id = 8; // Default: Physical Weapon (Group 8)
	if (.@wep_id == 40016) { // Twilight Staff (2H Staff)
		.@group_id = 9; // Magic (Group 9)
	} else if (.@wep_id == 40012 || .@wep_id == 40013 || .@wep_id == 40022 || .@wep_id == 40024 || .@wep_id == 40025) {
		// Twilight Kukri (Dagger), Edge (1H Sword), Codex (Book), Violin (Instrument), Whip (Whip)
		.@group_id = 10; // Mixed (Group 10)
	}

	// 2. Determine number of random option slots to apply (1 to 3 slots) matching C++ pc.cpp
	.@slots_to_apply = 1;
	if (rand(100) < 40) {
		.@slots_to_apply = 2;
		if (rand(100) < 15) {
			.@slots_to_apply = 3;
		}
	}

	// 3. Build Pool array matching exact C++ pools in pc.cpp
	deletearray .@pool[0], getarraysize(.@pool);
	.@pool_size = 0;

	if (.@group_id == 9) { // MAGICAL (Group 9)
		setarray .@s[0], 19, 151, 152, 155, 156, 168, 170, 171, 172;
		for (.@i = 0; .@i < 9; ++.@i) { .@pool[.@pool_size] = .@s[.@i]; .@pool_size++; }
		// Elemental Magic Dmg (57, 59, 61, 63, 65, 67, 69, 71, 73, 75)
		for (.@i = 57; .@i <= 75; .@i += 2) { .@pool[.@pool_size] = .@i; .@pool_size++; }
		// Magic Dmg vs Size (187, 188, 189)
		for (.@i = 187; .@i <= 189; ++.@i) { .@pool[.@pool_size] = .@i; .@pool_size++; }
	} else if (.@group_id == 10) { // MIXED (Group 10)
		setarray .@s[0], 16, 17, 18, 19, 24, 147, 148, 151, 152, 153, 154, 155, 156, 164, 168, 170, 171, 172;
		for (.@i = 0; .@i < 18; ++.@i) { .@pool[.@pool_size] = .@s[.@i]; .@pool_size++; }
		for (.@i = 37; .@i <= 55; .@i += 2) { .@pool[.@pool_size] = .@i; .@pool_size++; }
		for (.@i = 57; .@i <= 75; .@i += 2) { .@pool[.@pool_size] = .@i; .@pool_size++; }
		for (.@i = 97; .@i <= 106; ++.@i) { .@pool[.@pool_size] = .@i; .@pool_size++; }
		for (.@i = 157; .@i <= 159; ++.@i) { .@pool[.@pool_size] = .@i; .@pool_size++; }
		for (.@i = 187; .@i <= 189; ++.@i) { .@pool[.@pool_size] = .@i; .@pool_size++; }
	} else { // PHYSICAL (Group 8)
		setarray .@s[0], 16, 17, 18, 24, 147, 148, 153, 154, 164, 171, 172;
		for (.@i = 0; .@i < 11; ++.@i) { .@pool[.@pool_size] = .@s[.@i]; .@pool_size++; }
		// Elemental Phys Dmg (37, 39, 41, 43, 45, 47, 49, 51, 53, 55)
		for (.@i = 37; .@i <= 55; .@i += 2) { .@pool[.@pool_size] = .@i; .@pool_size++; }
		// Phys Dmg vs Race (97-106)
		for (.@i = 97; .@i <= 106; ++.@i) { .@pool[.@pool_size] = .@i; .@pool_size++; }
		// Phys Dmg vs Size (157-159)
		for (.@i = 157; .@i <= 159; ++.@i) { .@pool[.@pool_size] = .@i; .@pool_size++; }
	}

	// 4. Semaphores (Conflict Checks matching C++ pc.cpp)
	.@applied = 0;
	.@has_phys_ele = 0; .@has_mag_ele = 0; .@has_race_dmg = 0;
	.@has_size_phys = 0; .@has_size_mag = 0;

	while (.@applied < .@slots_to_apply && .@pool_size > 0) {
		.@target_idx = rand(.@pool_size);
		.@id_opt = .@pool[.@target_idx];

		// Remove element from pool
		.@pool[.@target_idx] = .@pool[.@pool_size - 1];
		.@pool_size--;

		// Conflict checks (semÃ¡foros)
		.@conflict = 0;
		if (.@id_opt >= 37 && .@id_opt <= 55) {
			if (.@has_phys_ele) .@conflict = 1; else .@has_phys_ele = 1;
		} else if (.@id_opt >= 57 && .@id_opt <= 75) {
			if (.@has_mag_ele) .@conflict = 1; else .@has_mag_ele = 1;
		} else if (.@id_opt >= 97 && .@id_opt <= 106) {
			if (.@has_race_dmg) .@conflict = 1; else .@has_race_dmg = 1;
		} else if (.@id_opt >= 157 && .@id_opt <= 159) {
			if (.@has_size_phys) .@conflict = 1; else .@has_size_phys = 1;
		} else if (.@id_opt >= 187 && .@id_opt <= 189) {
			if (.@has_size_mag) .@conflict = 1; else .@has_size_mag = 1;
		}

		if (.@conflict) continue;

		// Calculate exact values matching C++ pc.cpp formulas for Weapon Level 4
		.@val = 1;
		switch(.@id_opt) {
			case 17: .@val = rand(1, 25); break; // ATK (cap 25 for wlv 4)
			case 19: .@val = rand(1, 35); break; // MATK (cap 35 for wlv 4)
			case 18: .@val = rand(1, 15); break; // HIT (cap 15)
			case 16: case 24: case 170: case 171: .@val = rand(1, 10); break; // ASPD %, CRIT (cap 10 for wlv 4)
			case 153: case 154: .@val = rand(1, 20); break; // Long/Short Range Dmg % (cap 20 for wlv 4)
			case 155: case 156: .@val = rand(1, 15); break; // Cast time (cap 15)
			case 168: .@val = rand(1, 20); break; // Magic Dmg % (cap 20 for wlv 4)
			case 147: case 148: case 151: case 152: .@val = rand(1, 5); break; // Def/Mdef Ignore % (cap 5)
			case 187: case 188: case 189: .@val = rand(1, 10); break; // Magic Dmg vs Size (cap 10 for wlv 4)
			default:
				if ((.@id_opt >= 37 && .@id_opt <= 55) || (.@id_opt >= 57 && .@id_opt <= 75) || (.@id_opt >= 97 && .@id_opt <= 106) || (.@id_opt >= 157 && .@id_opt <= 159) || .@id_opt == 164 || .@id_opt == 172) {
					.@val = rand(1, 15); // cap 15 for wlv 4
				} else {
					.@val = rand(1, 10);
				}
				break;
		}

		setitemoption .@inv_idx, .@applied, .@id_opt, .@val;
		.@applied++;
	}

	// 5. Appending Durability Option (ID 243, Value .@durability) matching pc_set_item_durability
	if (.@applied < 5) {
		setitemoption .@inv_idx, .@applied, 243, .@durability;
	}

	return;
}
