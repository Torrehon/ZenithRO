// ==========================================================
// Sistema de Ciclo Ambiental Dinámico y Mana Engine
// Estados Globales: Día (0), Noche (1), Lluvia de Estrellas (2)
// ==========================================================

-	script	EnvCycleManager	-1,{
OnInit:
	if ($StarShowerChance == 0) set $StarShowerChance, 20; 
	
	// Valores base por seguridad al arrancar
	set $Global_EnvState, 0;
	set $ManaEvent_Active, 0;
	set $ManaEvent_Reloaded, 0;
	setbattleflag "base_exp_rate", 500;
	setbattleflag "job_exp_rate", 500;
	end;

// ==========================================================
// Gatillos Manuales (Llamados desde el NPC Interruptor)
// ==========================================================
OnManualStart:
	set $StarShowerChance, 100; // Forzamos el éxito
	goto L_TriggerCycle;
	end;

OnManualStop:
	set $ManaEvent_Active, 0;
	set $ManaEvent_Reloaded, 0;
	setbattleflag "base_exp_rate", 500;
	setbattleflag "job_exp_rate", 500;
	
	// Si estaba activa la lluvia, limpiamos el clima y lo hacemos de día por defecto
	if ($Global_EnvState == 2) {
		set $Global_EnvState, 0;
		day;
		atcommand "@clearweather";
	}
	
	// Recarga de mobs (Requiere RID)
	if (getusers(1) > 0) {
		if (query_sql("SELECT char_id FROM `char` WHERE online=1 LIMIT 1", .@target_ids)) {
			attachrid(.@target_ids[0]);
			atcommand "@reloadmobdb";
		}
	}
	
	announce "The mana flow has been manually stabilized. EXP rates are back to 5.00x.", bc_all|bc_blue;
	end;

// ==========================================================
// Gatillo Automático (Cada 3 horas)
// ==========================================================
OnHour00:OnHour03:OnHour06:OnHour09:OnHour12:OnHour15:OnHour18:OnHour21:
L_TriggerCycle: 

	set .@roll, rand(1, 100);

	if (.@roll <= $StarShowerChance) {
		// ==============================================
		// ¡ÉXITO! LLUVIA DE ESTRELLAS (Estado 2)
		// ==============================================
		set $Global_EnvState, 2;
		set $StarShowerChance, 20; 
		
		night;
		atcommand "@leaves"; 
		
		// MANA ENGINE: Calculo Aleatorio 6x - 8x
		set .@new_rate, rand(600, 800);
		set .@int, .@new_rate / 100;
		set .@dec, .@new_rate % 100;

		setbattleflag "base_exp_rate", .@new_rate;
		setbattleflag "job_exp_rate", .@new_rate;

		set $ManaEvent_Active, 1;
		set $ManaEvent_Reloaded, 0;
		
		// Recarga de Mobs (Requiere RID)
		if (getusers(1) > 0) {
			if (query_sql("SELECT char_id FROM `char` WHERE online=1 LIMIT 1", .@target_ids)) {
				attachrid(.@target_ids[0]);
				atcommand "@reloadmobdb";
				set $ManaEvent_Reloaded, 1;
			}
		}

		announce "A breathtaking meteor shower illuminates the dark sky! Mana flow burst, EXP bonus " + .@int + "." + .@dec + "x active.", bc_all|bc_blue;

	} else {
		// ==============================================
		// FALLO: CICLO NORMAL (Aumentamos probabilidad)
		// ==============================================
		set $StarShowerChance, $StarShowerChance + 10;
		
		// Apagamos el motor de exp si estaba encendido
		if ($ManaEvent_Active == 1) {
			set $ManaEvent_Active, 0;
			set $ManaEvent_Reloaded, 0;
			setbattleflag "base_exp_rate", 500;
			setbattleflag "job_exp_rate", 500;
			
			if (getusers(1) > 0) {
				if (query_sql("SELECT char_id FROM `char` WHERE online=1 LIMIT 1", .@target_ids)) {
					attachrid(.@target_ids[0]);
					atcommand "@reloadmobdb";
				}
			}
		}

		// Alternamos entre Día y Noche
		if ($Global_EnvState == 0 || $Global_EnvState == 2) {
			set $Global_EnvState, 1;
			night;
			atcommand "@clearweather";
			announce "The moon takes over as a quiet night falls.", bc_all|bc_blue;
		} else {
			set $Global_EnvState, 0;
			day;
			atcommand "@clearweather";
			announce "The sun rises, bringing a warm new day.", bc_all|bc_yellow;
		}
	}
	end;

OnPCLoginEvent:
	if ($ManaEvent_Active == 1 && $ManaEvent_Reloaded == 0) {
		atcommand "@reloadmobdb";
		set $ManaEvent_Reloaded, 1;
	}
	end;
}

// ==========================================================
// NPC CONTROL (PRONTERA 155, 180)
// ==========================================================
prontera,155,180,5	script	Mana Control	122,{
	mes "[Environment & Mana Control]";
	mes "Hello. (Testing Mode Active)";
	mes "Global State: ^0000FF" + (($Global_EnvState == 2) ? "Star Shower" : (($Global_EnvState == 1) ? "Night" : "Day")) + "^000000";
	mes "Next Shower Chance: ^FF0000" + $StarShowerChance + "%^000000";
	mes "Mana Event: " + (($ManaEvent_Active) ? "^00FF00Active^000000" : "^FF0000Inactive^000000");
	next;
	
	switch(select("Force Star Shower (6x-8x):Stop Event:Cancel")) {
		case 1:
			mes "[Mana Control]";
			mes "Triggering global event and reloading database...";
			close2;
			donpcevent "EnvCycleManager::OnManualStart";
			end;
		
		case 2:
			mes "[Mana Control]";
			mes "Restoring rates to 5.00x and clearing weather...";
			close2;
			donpcevent "EnvCycleManager::OnManualStop";
			end;

		case 3:
			close;
			end;
	}
}
