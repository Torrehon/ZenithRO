//===== rAthena Script =======================================
//= Stylist NPC (Pre-Renewal Compatible)
//===== Description: =========================================
//= Custom interactive Stylist for Pre-RE servers.
//= Allows players to customize Hair Style, Hair Color,
//= and Cloth Color using in-game preview & custom menu.
//============================================================

prt_in,31,114,4	script	Royal Stylist#custom_stylist	4_F_02,{
	mes "[Royal Stylist]";
	mes "Hello, " + strcharinfo(0) + "! Welcome to our beauty salon.";
	mes "What would you like to customize today?";
	next;

	setarray .@Styles[1],
		getbattleflag("max_cloth_color"),
		getbattleflag("max_hair_style"),
		getbattleflag("max_hair_color");
	setarray .@Look[1],
		LOOK_CLOTHES_COLOR,
		LOOK_HAIR,
		LOOK_HAIR_COLOR;

	set .@s, select(" ~ Clothes Color: ~ Hair Style: ~ Hair Color: ~ Cancel");
	if (.@s == 4) {
		mes "[Royal Stylist]";
		mes "Come back anytime you want to refresh your style!";
		close;
	}

	set .@Revert, getlook(.@Look[.@s]);
	set .@Style, (.@Revert ? .@Revert : 1);

	while(1) {
		setlook .@Look[.@s], .@Style;
		message strcharinfo(0), "Current style: #" + .@Style + " (Max: " + .@Styles[.@s] + ")";
		set .@NextStyle, ((.@Style != .@Styles[.@s]) ? .@Style + 1 : 1);
		set .@PrevStyle, ((.@Style != 1) ? .@Style - 1 : .@Styles[.@s]);
		set .@menu$, " ~ Next (^0055FF#" + .@NextStyle + "^000000)"
		           + ": ~ Previous (^0055FF#" + .@PrevStyle + "^000000)"
		           + ": ~ Jump to style number..."
		           + ": ~ Random Style"
		           + ": ~ ^008800Confirm this Style^000000"
		           + ": ~ ^FF0000Revert and Exit^000000";
		
		switch(select(.@menu$)) {
		case 1:
			set .@Style, .@NextStyle;
			break;
		case 2:
			set .@Style, .@PrevStyle;
			break;
		case 3:
			message strcharinfo(0), "Enter a number between 1 and " + .@Styles[.@s] + ".";
			input .@Style, 1, .@Styles[.@s];
			break;
		case 4:
			set .@Style, rand(1, .@Styles[.@s]);
			break;
		case 5:
			mes "[Royal Stylist]";
			mes "Great choice! You look amazing with your new look.";
			close;
		case 6:
			setlook .@Look[.@s], .@Revert;
			mes "[Royal Stylist]";
			mes "Reverted to your original style. See you next time!";
			close;
		}
	}
}
