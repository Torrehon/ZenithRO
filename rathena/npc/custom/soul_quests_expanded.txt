// Valkyrie Kara in Chunin Sanctuary (1@exse 43, 23)
1@exse,43,23,4	script	Valkyrie#exse	811,{
	mes "[Valkyrie]";
	mes "Greetings, shadow warrior. I am Valkyrie Kara, guardian of the sacred realm.";
	mes "You stand within the ^0000FFChunin Sanctuary^000000, where the spirits of ancient Ninja champions rest in eternal honor.";
	next;
	mes "[Valkyrie]";
	mes "Seek out the Chunin Einherjar of your order, offer your tribute, and claim their ancient Hero Soul.";
	next;
	mes "[Valkyrie]";
	mes "How may I guide your path today?";
	next;
	switch(select("Re-align Twilight Weapon Options (Reroll):Return to Geffen Field (gef_fild07):Remain in Sanctuary")) {
		case 1:
			callfunc "F_ValkyrieReroll";
			end;
		case 2:
			mes "[Valkyrie]";
			mes "May the shadows protect your path.";
			close2;
			warp "gef_fild07",173,246;
			end;
		default:
			mes "[Valkyrie]";
			mes "Go forth and speak with the Chunin Einherjar.";
			close;
	}
}

// Chunin Einherjar (1@exse 42, 65) - Sprite 4306 [Skills: 1055, 1056]
1@exse,42,65,4	script	Chunin Einherjar	4304,{
	mes "[Chunin Einherjar]";
	mes "Step from the shadows, traveler. I am Master Shadow-Blade Raizo of Amatsu.";
	mes "In life, I mastered the secret arts of the Ninjutsu scrolls and the swift lethal edge of the Ninja Guild.";
	next;
	if (BaseJob != Job_Ninja) {
		mes "[Chunin Einherjar]";
		mes "However, my secret shadow scrolls belong only to Ninja and Chunin warriors. Walk your own path with honor.";
		close;
	}
		if (getskilllv(1056) > 0 && getskilllv(1032) == 0) {
		skill 1032, 1, SKILL_PERM_GRANT;
		specialeffect2 EF_ANGEL;
		mes "[Chunin Einherjar]";
		mes "Ah! You possess the ^0000FFSoul of the Kuji^000000! I have granted you your ^0000FFDual Wield^000000 skill!";
		close;
	}
	if (JobLevel < 25) {
		mes "[Chunin Einherjar]";
		mes "Your footfalls are still too loud and your Ninjutsu unpolished. Return when you reach ^0000FFJob Level 25^000000.";
		close;
	}
	if ((SOUL_QUEST1_STEP >= 4 || checkquest(90004) >= 1 || checkquest(90005) >= 1) && !F_HasSoulSkill(1055, 1056)) {
		mes "[Chunin Einherjar]";
		mes "Greetings, shadow brother. I sense that the dark void attempted to sever your link with our ancestral Hero Soul...";
		mes "Fear not! Your heroic deeds remain recorded in Valhalla. Select your Hero Soul to restore it at no cost:";
		next;
		switch(select("Soul of the Kensei:Soul of the Kuji:Cancel")) {
			case 1:
				skill 1055, 1, SKILL_PERM_GRANT;
				mes "[Chunin Einherjar]";
				mes "Your ^0000FFSoul of the Kensei^000000 has been restored!";
				close;
			case 2:
				skill 1056, 1, SKILL_PERM_GRANT;
				skill 1032, 1, SKILL_PERM_GRANT;
				mes "[Chunin Einherjar]";
				mes "Your ^0000FFSoul of the Kuji^000000 and ^0000FFDual Wield^000000 have been restored!";
				close;
			default:
				close;
		}
	}
	if (SOUL_QUEST1_STEP >= 5) {
		mes "[Chunin Einherjar]";
		mes "Greetings, shadow brother. You have already awakened the Star Fragment into your Twilight Weapon!";
		mes "May the darkness protect your path.";
		close;
	}
	if (F_HasSoulSkill(1055, 1056)) {
		mes "[Chunin Einherjar]";
		mes "Greetings, shadow brother. You carry the ancient shadow soul of our guild within your spirit.";
		next;
		if (select("Evolve Dawn Weapon:Leave") == 1) {
			callfunc "F_EinherjarWeaponMenu";
			end;
		}
		close;
	}


	mes "[Chunin Einherjar]";
	mes "To resonate with the ancestral shadow soul of the Ninja Guild, your blade requires a vessel of raw heroic power.";
	mes "Offer 1 ^0000FFHeroic Soul Essence^000000 to ignite the ancient Ninjutsu scrolls.";
	next;
	if (countitem(50020) < 1) {
		mes "[Chunin Einherjar]";
		mes "Alas, you carry no ^0000FFHeroic Soul Essence^000000 (Item ID 50020).";
		mes "Ancient fragments of hero souls lie scattered across Midgard. Wild beasts that absorbed these fragments became monsters far out of the ordinary.";
		mes "Hunt down these unusual creatures across the lands to claim their Heroic Soul Essence!";
		close;
	}

	mes "[Chunin Einherjar]";
	mes "Choose the hero soul that shall empower your shadow discipline:";
	mes "- ^0000FFSoul of the Kensei^000000: Sword and martial specialist focusing on deadly melee strikes and physical shadow arts.";
	mes "- ^0000FFSoul of the Kuji^000000: Elemental Ninjutsu spell specialist focusing on catastrophic magic and elemental incantations.";
	next;
	switch(select("Soul of the Kensei:Soul of the Kuji:Cancel")) {
		case 1:
			delitem 50020, 1;
			skill 1055, 1, SKILL_PERM_GRANT;
			SOUL_QUEST1_STEP = 4;
			if (checkquest(90004) == 1) completequest 90004;
			specialeffect2 EF_ANGEL;
			mes "[Chunin Einherjar]";
			mes "You have received the ^0000FFSoul of the Kensei^000000. May your blade strike unseen in the dark!";
			close;
		case 2:
			delitem 50020, 1;
			skill 1056, 1, SKILL_PERM_GRANT; // NJ_KUJISOUL
			skill 1032, 1, SKILL_PERM_GRANT; // NJ_DUAL (Dual Wield)
			SOUL_QUEST1_STEP = 4;
			if (checkquest(90004) == 1) completequest 90004;
			specialeffect2 EF_ANGEL;
			mes "[Chunin Einherjar]";
			mes "You have received the ^0000FFSoul of the Kuji^000000 and acquired ^0000FFDual Wield^000000!";
			mes "Command the primordial Ninjutsu elements and wield dual daggers in the shadows!";
			close;
		default:
			close;
	}
OnInit:
	.@gid = getnpcid(0);
	setunitdata .@gid, UNPC_SEX, 1;
	setunitdata .@gid, UNPC_HAIRSTYLE, 16;
	setunitdata .@gid, UNPC_HAIRCOLOR, 4;
	setunitdata .@gid, UNPC_CLOTHCOLOR, 3;
	end;
}
