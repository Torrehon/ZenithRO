//===== rAthena Script =======================================
//= Schwarzwald-Arunafeltz Border Checkpoint & Safety Net
//===== Description: =========================================
//= Closes land passage between Lighthalzen Field and Rachel,
//= and provides a safety ejector for restricted post-Ep 10 maps.
//============================================================

// Border Guards at Lighthalzen Field 01 canyon pass
//============================================================
lhz_fild01,293,378,3	script	Schwarzwald Border Guard::LhzBorderGuard	105,{
	mes "[Border Guard]";
	mes "Halt, traveler!";
	mes "Beyond this mountain gorge lies the frontier with the Holy Ground of Arunafeltz.";
	next;
	mes "[Border Guard]";
	mes "By decree of the Schwarzwald Republic Council, all civilian transit across this border has been suspended indefinitely due to heightened diplomatic tensions.";
	close;
}

lhz_fild01,299,378,5	duplicate(LhzBorderGuard)	Schwarzwald Border Guard#2	105

// Invisible Barrier / Step-back Trigger
//============================================================
lhz_fild01,296,379,0	script	#SchwarzwaldBorder	-1,3,1,{
	end;

OnTouch:
	if (getgmlevel() >= 60) end;
	message strcharinfo(0), "Border Guard: Halt! Civilian travel into the Arunafeltz territory is strictly prohibited.";
	unitwalk getcharid(3), 296, 374;
	end;
}

// Safety Perimeter Net for Post-Episode 10 Maps
//============================================================
-	script	ArunafeltzPerimeterNet	-1,{
	end;

OnPCLoadMapEvent:
	if (getgmlevel() >= 60) end;
	.@current_map$ = strcharinfo(3);
	for (.@i = 0; .@i < .restricted_count; .@i++) {
		if (.@current_map$ == .restricted_maps$[.@i]) {
			warp "yuno", 158, 125;
			dispbottom "The territory of Arunafeltz and subsequent regions are currently closed to all travelers.";
			end;
		}
	}
	end;

OnInit:
	setarray .restricted_maps$[0],
		// Rachel Town & Interiors
		"rachel", "ra_in01", "ra_temple", "ra_temin", "ra_temsky", "que_san04",
		// Rachel Fields
		"ra_fild01", "ra_fild02", "ra_fild03", "ra_fild04", "ra_fild05", "ra_fild06",
		"ra_fild07", "ra_fild08", "ra_fild09", "ra_fild10", "ra_fild11", "ra_fild12", "ra_fild13",
		// Rachel Dungeons
		"ice_dun01", "ice_dun02", "ice_dun03",
		"ra_san01", "ra_san02", "ra_san03", "ra_san04", "ra_san05",
		// Veins Town & Fields
		"veins", "ve_in",
		"ve_fild01", "ve_fild02", "ve_fild03", "ve_fild04", "ve_fild05", "ve_fild06", "ve_fild07",
		// Thor Volcano
		"thor_v01", "thor_v02", "thor_v03", "thor_camp",
		// Nameless Island & Abbey
		"nameless_n", "nameless_i", "nameless_in",
		"abbey01", "abbey02", "abbey03",
		// Ash Vacuum / Midgard Camp / Other Subsequent Regions
		"mid_camp", "mid_campin", "manuk", "man_in01", "splendide", "spl_in01", "spl_in02",
		"man_fild01", "man_fild02", "man_fild03", "spl_fild01", "spl_fild02", "spl_fild03",
		"nyd_dun01", "nyd_dun02", "dicastes01", "dicastes02", "mora", "eclage", "malaya", "dewata",
		"brasilis", "bra_fild01", "bra_dun01", "moscovia", "mosk_fild01", "mosk_fild02", "mosk_dun01", "mosk_dun02", "mosk_dun03";

	.restricted_count = getarraysize(.restricted_maps$);
	for (.@i = 0; .@i < .restricted_count; .@i++) {
		setmapflag .restricted_maps$[.@i], mf_loadevent;
		setmapflag .restricted_maps$[.@i], mf_nomemo;
	}
	end;
}
