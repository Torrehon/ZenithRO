// =====================================================
// Custom Hunting System - Payon

// -----------------------------------------------------
// 1. Hunting Board - Payon (Sprite 857)
// -----------------------------------------------------
payon,188,130,4	script	Hunting Board#pay	857,{
	.@today = gettime(DT_YYYYMMDD);

	while (1) {
		mes "[^0055FF" + .BoardTitle$ + "^000000]";
		mes "Greetings, hunter. Select a contract tier to inspect today's missions in Payon forests and caves.";
		mes "You may take on multiple contracts at once.";
		next;

		// Check how many completed contracts are ready to turn in
		.@ready_count = 0;
		for (.@i = 1; .@i <= .TotalMobs; .@i++) {
			if (isbegin_quest(.Mob_QID[.@i]) == 1 && checkquest(.Mob_QID[.@i], HUNTING) == 2) {
				.@ready_count++;
			}
		}

		setarray .@Tiers$[1], "^00AA00Tier D (1-25)^000000", "^0055FFTier C (25-50)^000000", "^FF8800Tier B (50-75)^000000", "^FF0000Tier A (75-99)^000000", "^AA00AATier S (Minibosses)^000000";

		if (.@ready_count > 0) {
			.@menu$ = "^00AA00[Claim All] Completed Rewards (" + .@ready_count + " Ready)^000000:";
		} else {
			.@menu$ = "";
		}
		.@menu$ = .@menu$ + .@Tiers$[1] + ":" + .@Tiers$[2] + ":" + .@Tiers$[3] + ":" + .@Tiers$[4] + ":" + .@Tiers$[5] + ":^FF0000Manage Active Contracts^000000:Close";

		.@sel = select(.@menu$);

		// Handle Claim All Completed
		if (.@ready_count > 0) {
			if (.@sel == 1) {
				.@total_base = 0;
				.@total_job = 0;
				.@total_coins = 0;
				.@claimed_count = 0;

				for (.@i = 1; .@i <= .TotalMobs; .@i++) {
					if (isbegin_quest(.Mob_QID[.@i]) == 1 && checkquest(.Mob_QID[.@i], HUNTING) == 2) {
						if (.Mob_Tier[.@i] == 5) {
							.@base = (getmonsterinfo(.Mob_ID[.@i], MOB_BASEEXP) * 150 / 100);
							.@job  = (getmonsterinfo(.Mob_ID[.@i], MOB_JOBEXP) * 150 / 100);
						} else {
							.@base = ((getmonsterinfo(.Mob_ID[.@i], MOB_BASEEXP) * .Mob_Req[.@i]) * 75 / 100);
							.@job  = ((getmonsterinfo(.Mob_ID[.@i], MOB_JOBEXP) * .Mob_Req[.@i]) * 75 / 100);
						}
						.@total_base += .@base;
						.@total_job += .@job;
						.@total_coins += .Mob_Coins[.@i];
						erasequest .Mob_QID[.@i];
						setd "Hunt_CD_" + .Mob_ID[.@i], .@today;
						.@claimed_count++;
					}
				}

				getexp .@total_base, .@total_job;
				getitem .CoinID, .@total_coins;

				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "Outstanding achievement! You have completed ^00AA00" + .@claimed_count + "^000000 hunting contract(s).";
				mes "^00AA00You received a total of " + .@total_coins + "x " + .CoinName$ + "(s) and extra Experience!^000000";
				specialeffect2 EF_MONEY;
				next;
				continue;
			}
			// Adjust selected tier offset
			.@selTier = .@sel - 1;
		} else {
			.@selTier = .@sel;
		}

		if (.@selTier == 7) close;

		if (.@selTier == 6) {
			deletearray .@active_qids[0];
			.@active_count = 0;
			.@ready_in_active = 0;
			for (.@i = 1; .@i <= .TotalMobs; .@i++) {
				if (isbegin_quest(.Mob_QID[.@i]) == 1) {
					.@active_qids[.@active_count] = .@i;
					.@active_count++;
					if (checkquest(.Mob_QID[.@i], HUNTING) == 2) {
						.@ready_in_active++;
					}
				}
			}

			if (.@active_count == 0) {
				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "You currently have no active hunting contracts from this board.";
				next;
				continue;
			}

			mes "[^0055FF" + .BoardTitle$ + "^000000]";
			mes "Select an active contract to inspect or abandon:";
			next;

			.@ab_menu$ = "";
			if (.@ready_in_active > 0) {
				.@ab_menu$ = "^00AA00[Claim All] Completed Contracts (" + .@ready_in_active + ")^000000:";
			}
			for (.@i = 0; .@i < .@active_count; .@i++) {
				.@m = .@active_qids[.@i];
				if (checkquest(.Mob_QID[.@m], HUNTING) == 2) {
					.@ab_menu$ = .@ab_menu$ + "^00AA00[Ready to Report] " + .Mob_Name$[.@m] + "^000000:";
				} else {
					.@ab_menu$ = .@ab_menu$ + "^FF0000[Abandon] " + .Mob_Name$[.@m] + "^000000:";
				}
			}
			.@ab_menu$ = .@ab_menu$ + "^FF0000Abandon ALL Active Contracts^000000:Back to Main Menu";

			.@ab_sel = select(.@ab_menu$);

			// Handle Claim All inside Manage Contracts
			if (.@ready_in_active > 0 && .@ab_sel == 1) {
				.@total_base = 0;
				.@total_job = 0;
				.@total_coins = 0;
				.@claimed_count = 0;

				for (.@i = 0; .@i < .@active_count; .@i++) {
					.@m = .@active_qids[.@i];
					if (checkquest(.Mob_QID[.@m], HUNTING) == 2) {
						if (.Mob_Tier[.@m] == 5) {
							.@base = (getmonsterinfo(.Mob_ID[.@m], MOB_BASEEXP) * 150 / 100);
							.@job  = (getmonsterinfo(.Mob_ID[.@m], MOB_JOBEXP) * 150 / 100);
						} else {
							.@base = ((getmonsterinfo(.Mob_ID[.@m], MOB_BASEEXP) * .Mob_Req[.@m]) * 75 / 100);
							.@job  = ((getmonsterinfo(.Mob_ID[.@m], MOB_JOBEXP) * .Mob_Req[.@m]) * 75 / 100);
						}
						.@total_base += .@base;
						.@total_job += .@job;
						.@total_coins += .Mob_Coins[.@m];
						erasequest .Mob_QID[.@m];
						setd "Hunt_CD_" + .Mob_ID[.@m], .@today;
						.@claimed_count++;
					}
				}

				getexp .@total_base, .@total_job;
				getitem .CoinID, .@total_coins;

				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "Outstanding achievement! You have completed ^00AA00" + .@claimed_count + "^000000 hunting contract(s).";
				mes "^00AA00You received a total of " + .@total_coins + "x " + .CoinName$ + "(s) and extra Experience!^000000";
				specialeffect2 EF_MONEY;
				next;
				continue;
			}

			// Adjust offset if Claim All was present
			.@ab_choice = (.@ready_in_active > 0) ? (.@ab_sel - 1) : .@ab_sel;

			if (.@ab_choice == .@active_count + 2) continue;

			if (.@ab_choice == .@active_count + 1) {
				for (.@i = 0; .@i < .@active_count; .@i++) {
					.@m = .@active_qids[.@i];
					erasequest .Mob_QID[.@m];
				}
				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "All active contracts have been abandoned.";
				next;
				continue;
			}

			.@m = .@active_qids[.@ab_choice - 1];
			// If it was ready to report, turn it in individually!
			if (checkquest(.Mob_QID[.@m], HUNTING) == 2) {
				if (.Mob_Tier[.@m] == 5) {
					.@base = (getmonsterinfo(.Mob_ID[.@m], MOB_BASEEXP) * 150 / 100);
					.@job  = (getmonsterinfo(.Mob_ID[.@m], MOB_JOBEXP) * 150 / 100);
				} else {
					.@base = ((getmonsterinfo(.Mob_ID[.@m], MOB_BASEEXP) * .Mob_Req[.@m]) * 75 / 100);
					.@job  = ((getmonsterinfo(.Mob_ID[.@m], MOB_JOBEXP) * .Mob_Req[.@m]) * 75 / 100);
				}

				getexp .@base, .@job;
				getitem .CoinID, .Mob_Coins[.@m];
				erasequest .Mob_QID[.@m];
				setd "Hunt_CD_" + .Mob_ID[.@m], .@today;

				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "Excellent work! Contract for ^00AA00" + .Mob_Name$[.@m] + "^000000 completed.";
				mes "^00AA00You receive " + .Mob_Coins[.@m] + "x " + .CoinName$ + "(s) and extra Experience.^000000";
				specialeffect2 EF_MONEY;
				next;
				continue;
			}

			erasequest .Mob_QID[.@m];
			mes "[^0055FF" + .BoardTitle$ + "^000000]";
			mes "Contract for ^FF0000" + .Mob_Name$[.@m] + "^000000 abandoned.";
			next;
			continue;
		}

		while (1) {
			mes "[^0055FF" + .BoardTitle$ + "^000000]";
			mes "Select a contract from " + .@Tiers$[.@selTier] + ":";
			next;

			deletearray .@mob_idx[0];
			.@count = 0;
			.@avail_count = 0;
			for (.@i = 1; .@i <= .TotalMobs; .@i++) {
				if (.Mob_Tier[.@i] == .@selTier) {
					.@mob_idx[.@count] = .@i;
					.@count++;
					.@qid = .Mob_QID[.@i];
					.@cd = getd("Hunt_CD_" + .Mob_ID[.@i]);
					if (.@cd != .@today && isbegin_quest(.@qid) == 0) {
						.@avail_count++;
					}
				}
			}

			if (.@count == 0) {
				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "Coming soon...";
				next;
				break;
			}

			// Build menu dynamically based on quest_db state
			.@menu$ = "";
			if (.@avail_count > 0) {
				.@menu$ = "^00AA00[Accept All Available (" + .@avail_count + ")]^000000:";
			}

			for (.@i = 0; .@i < .@count; .@i++) {
				.@m = .@mob_idx[.@i];
				.@qid = .Mob_QID[.@m];
				.@cd = getd("Hunt_CD_" + .Mob_ID[.@m]);
				.@st = isbegin_quest(.@qid);
				.@hunt = checkquest(.@qid, HUNTING);

				if (.@cd == .@today) {
					.@menu$ = .@menu$ + "^777777[Done] " + .Mob_Name$[.@m] + " (" + .Mob_Req[.@m] + ")^000000:";
				} else if (.@st == 1 && .@hunt == 2) {
					.@menu$ = .@menu$ + "^00AA00[Report] " + .Mob_Name$[.@m] + " (" + .Mob_Req[.@m] + ")^000000:";
				} else if (.@st == 1) {
					.@menu$ = .@menu$ + "^0055FF[In Progress] " + .Mob_Name$[.@m] + " (" + .Mob_Req[.@m] + ")^000000:";
				} else {
					.@menu$ = .@menu$ + "[Accept] " + .Mob_Name$[.@m] + " (" + .Mob_Req[.@m] + "):";
				}
			}
			.@menu$ = .@menu$ + "Back to Tiers";

			.@choice = select(.@menu$);

			// Handle Accept All option if present
			if (.@avail_count > 0 && .@choice == 1) {
				.@accepted_count = 0;
				for (.@i = 0; .@i < .@count; .@i++) {
					.@m = .@mob_idx[.@i];
					.@qid = .Mob_QID[.@m];
					.@cd = getd("Hunt_CD_" + .Mob_ID[.@m]);
					if (.@cd != .@today && isbegin_quest(.@qid) == 0) {
						setquest .@qid;
						.@accepted_count++;
					}
				}
				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "Successfully accepted ^00AA00" + .@accepted_count + "^000000 hunting contract(s) for " + .@Tiers$[.@selTier] + "!";
				mes "Track your hunting progress in the Quest Log (^0055FFAlt + U^000000).";
				next;
				continue;
			}

			// Handle Back button
			if (.@avail_count > 0 && .@choice == .@count + 2) {
				break;
			} else if (.@avail_count == 0 && .@choice == .@count + 1) {
				break;
			}

			// Get selected monster index
			.@m = .@mob_idx[(.@avail_count > 0 ? .@choice - 2 : .@choice - 1)];
			.@qid = .Mob_QID[.@m];
			.@cd = getd("Hunt_CD_" + .Mob_ID[.@m]);
			.@st = isbegin_quest(.@qid);
			.@hunt = checkquest(.@qid, HUNTING);

			// Check if already completed today
			if (.@cd == .@today) {
				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "Target: ^0055FF" + .Mob_Name$[.@m] + "^000000";
				mes "Status: ^777777Completed for today^000000";
				mes " ";
				mes "You have already completed this contract today.";
				mes "Check back tomorrow for a new contract.";
				next;
				continue;
			}

			// Check if ready to report or in progress
			if (.@st == 1 && .@hunt == 2) {
				// Quest completed!
				if (.Mob_Tier[.@m] == 5) {
					.@base = (getmonsterinfo(.Mob_ID[.@m], MOB_BASEEXP) * 150 / 100);
					.@job  = (getmonsterinfo(.Mob_ID[.@m], MOB_JOBEXP) * 150 / 100);
				} else {
					.@base = ((getmonsterinfo(.Mob_ID[.@m], MOB_BASEEXP) * .Mob_Req[.@m]) * 75 / 100);
					.@job  = ((getmonsterinfo(.Mob_ID[.@m], MOB_JOBEXP) * .Mob_Req[.@m]) * 75 / 100);
				}

				getexp .@base, .@job;
				getitem .CoinID, .Mob_Coins[.@m];
				erasequest .@qid;
				setd "Hunt_CD_" + .Mob_ID[.@m], .@today;

				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "Excellent work! Contract for ^00AA00" + .Mob_Name$[.@m] + "^000000 completed.";
				mes "^00AA00You receive " + .Mob_Coins[.@m] + "x " + .CoinName$ + "(s) and extra Experience.^000000";
				specialeffect2 EF_MONEY;
				next;
				continue;
			} else if (.@st == 1) {
				mes "Contract: ^0055FF" + .Mob_Name$[.@m] + "^000000";
				mes "Status: ^0055FFActive in Quest Log (Alt + U)^000000";
				next;
				if (select("Keep hunting:Abandon contract") == 2) {
					erasequest .@qid;
					mes "Contract abandoned.";
					next;
				}
				continue;
			}

			// Confirmation Window with quest details
			mes "=== ^0055FFContract Details^000000 ===";
			mes "Target Monster: ^FF0000" + .Mob_Name$[.@m] + "^000000";
			mes "Required Kills: ^00AA00" + .Mob_Req[.@m] + "^000000";
			mes "Reward Coins: ^00AA00" + .Mob_Coins[.@m] + "x " + .CoinName$ + "^000000";
			if (.Mob_Tier[.@m] == 5) {
				mes "Bonus Experience: ^00AA00150% of Miniboss EXP^000000";
			} else {
				mes "Bonus Experience: ^00AA0075% of total hunting EXP^000000";
			}
			mes " ";
			mes "Do you wish to accept this hunting contract?";
			next;

			if (select("Accept Contract:Decline") == 2) {
				mes "[^0055FF" + .BoardTitle$ + "^000000]";
				mes "Contract declined.";
				next;
				continue;
			}

			setquest .@qid;
			mes "[^0055FF" + .BoardTitle$ + "^000000]";
			mes "Contract signed! Good luck hunting ^FF0000" + .Mob_Name$[.@m] + "^000000.";
			mes "You can view progress in your Quest Log (^0055FFAlt + U^000000).";
			next;
			continue;
		}
	}

	// Mob Data Structure
	setarray .Mob_Name$[1],

payon,186,130,4	script	Veteran Hunter#pay	727,{
	mes "[^0055FFVeteran Hunter^000000]";
	mes "Greetings, traveler. May the spirits of the forest guide your arrows and keep you safe from the restless dead.";
	mes "I reward those who help keep the mountain paths and dark caves safe from wild beasts.";
	mes "Do you wish to exchange your hard-earned Payon Coins for provisions?";
	mes " ";
	mes "You currently hold: ^00AA00" + countitem(50006) + " Payon Coins^000000.";
	next;

	if (select("Show me your wares:I must return to the wild") == 2) {
		mes "[^0055FFVeteran Hunter^000000]";
		mes "Walk in harmony with nature, hunter.";
		close;
	}

	callshop "payon_hunter_shop",1;
	npcshopattach "payon_hunter_shop";
	end;

OnBuyItem:
	.@size = getarraysize(@bought_nameid);
	for (.@i = 0; .@i < .@size; .@i++) {
		.@item_idx = inarray(.Shop_Data, @bought_nameid[.@i]);
		if (.@item_idx == -1) {
			dispbottom "System Error: Item not found in the supply line.";
			end;
		}
		.@total_cost += (.Shop_Data[.@item_idx + 1] * @bought_quantity[.@i]);
	}

	if (.@total_cost > countitem(50006)) {
		mes "[^0055FFVeteran Hunter^000000]";
		mes "I am afraid you lack the necessary Payon Coins for these supplies.";
		mes "Total cost: " + .@total_cost + " Coins.";
		close;
	}
	
	delitem 50006, .@total_cost;
	
	for (.@i = 0; .@i < .@size; .@i++) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
	}
	
	mes "[^0055FFVeteran Hunter^000000]";
	mes "A wise choice. May these provisions serve you well in the deep woods and dark caverns.";
	mes "Stay alert!";
	specialeffect2 EF_MONEY;
	close;

OnInit:
	setarray .Shop_Data[0], 
		13531, 3,    // Light Red Potion Box
		13532, 5,    // Light Orange Potion Box
		13533, 7,    // Light Yellow Potion Box
		13534, 10,   // Light White Potion Box
		12114, 2,    // Fire Converter Scroll
		12115, 2,    // Water Converter Scroll
		12116, 2,    // Earth Converter Scroll
		12117, 2,    // Wind Converter Scroll
		51008, 3,    // Shadow Converter Scroll
		51009, 3,    // Holy Converter Scroll
		51018, 2,
		51019, 2,
		51020, 1,
		19639, 50,   // Costume Sakkat
		19503, 50,   // Costume Munak Hat
		19687, 50,   // Costume Lush Rose
		19292, 200,  // Costume Wild Cat Knit Cap
		19536, 200,  // Costume Rose Corsage
		19543, 200,  // Costume Oliver Wolf Hood
		19545, 200,  // Costume Boy's Cap
		19598, 500,  // Costume Wandering Wolf King Helmet
		19642, 500,  // Costume Moonlight Flower Hat
		31415, 500;  // Costume Wanderer's Sakkat

	npcshopdelitem "payon_hunter_shop", 502; 
	for (.@i = 0; .@i < getarraysize(.Shop_Data); .@i += 2) {
		npcshopadditem "payon_hunter_shop", .Shop_Data[.@i], .Shop_Data[.@i+1];
	}
	end;
}
