				next;

// 3. Veteran Hunter Shop (Cave/Arche Flavor)
// -----------------------------------------------------
pay_arche,60,148,4	script	Veteran Hunter#arche	732,{
	mes "[^0055FFVeteran Hunter^000000]";
	mes "Well met, brave soul. The caves are treacherous, but your efforts keep the darkness from spilling out.";
	mes "I exchange Payon Coins for essential supplies. Do you need anything?";
	mes " ";
	mes "You currently hold: ^00AA00" + countitem(50006) + " Payon Coins^000000.";
	next;

	if (select("Show me your wares:I must return to the caves") == 2) {
		mes "[^0055FFVeteran Hunter^000000]";
		mes "Stay vigilant down there. The dead don't rest easily.";
		close;
	}

	callshop "arche_hunter_shop",1;
	npcshopattach "arche_hunter_shop";
	end;

OnBuyItem:
	.@size = getarraysize(@bought_nameid);
	for (.@i = 0; .@i < .@size; .@i++) {
		.@item_idx = inarray(.Shop_Data, @bought_nameid[.@i]);
		if (.@item_idx == -1) {
			dispbottom "System Error: Item not found in the supply line.";
			end;
		}
		.@total_cost += (.Shop_Data[.@item_idx + 1] * @bought_quantity[.@i]);
	}

	if (.@total_cost > countitem(50006)) {
		mes "[^0055FFVeteran Hunter^000000]";
		mes "I am afraid you lack the necessary Payon Coins for these supplies.";
		mes "Total cost: " + .@total_cost + " Coins.";
		close;
	}
	
	delitem 50006, .@total_cost;
	
	for (.@i = 0; .@i < .@size; .@i++) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
	}
	
	mes "[^0055FFVeteran Hunter^000000]";
	mes "A wise choice. May these provisions keep you alive in the depths.";
	mes "Good luck!";
	specialeffect2 EF_MONEY;
	close;

OnInit:
	setarray .Shop_Data[0], 
		502, 2,   // Red Potion (Ejemplo: Cuesta 2 monedas)
		501, 1;   // Red Herb (Ejemplo: Cuesta 1 moneda) 

	npcshopdelitem "arche_hunter_shop", 502; 
	for (.@i = 0; .@i < getarraysize(.Shop_Data); .@i += 2) {
		npcshopadditem "arche_hunter_shop", .Shop_Data[.@i], .Shop_Data[.@i+1];
	}
	end;
}
