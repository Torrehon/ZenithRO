// 3. Veteran Hunter Shop (Crusader Flavor)
// -----------------------------------------------------
prontera,141,172,4	script	Veteran Hunter	404,{
	mes "[^0055FFVeteran Hunter^000000]";
	mes "Greetings, citizen. May the Light protect you on your travels.";
	mes "I am tasked with rewarding those who take up arms to defend our lands from the encroaching beasts.";
	mes "Do you wish to exchange your hard-earned Prontera Coins for provisions?";
	mes " ";
	mes "You currently hold: ^00AA00" + countitem(50003) + " Prontera Coins^000000.";
	next;

	if (select("Show me your wares:I must return to my duties") == 2) {
		mes "[^0055FFVeteran Hunter^000000]";
		mes "Stand firm, and may your shield never falter.";
		close;
	}

	callshop "hunter_shop",1;
	npcshopattach "hunter_shop";
	end;

OnBuyItem:
	.@size = getarraysize(@bought_nameid);
	for (.@i = 0; .@i < .@size; .@i++) {
		.@item_idx = inarray(.Shop_Data, @bought_nameid[.@i]);
		if (.@item_idx == -1) {
			dispbottom "System Error: Item not found in the Crusader's supply line.";
			end;
		}
		.@total_cost += (.Shop_Data[.@item_idx + 1] * @bought_quantity[.@i]);
	}

	if (.@total_cost > countitem(50003)) {
		mes "[^0055FFVeteran Hunter^000000]";
		mes "I am afraid you lack the necessary Prontera Coins for these supplies, my friend.";
		mes "Total cost: " + .@total_cost + " Coins.";
		close;
	}
	
	delitem 50003, .@total_cost;
	
	for (.@i = 0; .@i < .@size; .@i++) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
	}
	
	mes "[^0055FFVeteran Hunter^000000]";
	mes "A wise choice. May these provisions serve you well on the battlefield.";
	mes "Stay vigilant!";
	specialeffect2 EF_MONEY;
	close;

OnInit:
	setarray .Shop_Data[0], 
		13531, 3,    // Light Red Potion Box
		13532, 5,    // Light Orange Potion Box
		13533, 7,    // Light Yellow Potion Box
		13534, 10,   // Light White Potion Box
		19526, 50,   // Costume Helm
		19630, 50,   // Costume Crown
		19832, 50,   // Costume Poring Hat
		19972, 200,  // Costume Crown of Guardian
		19673, 200,  // Costume Love Guard
		19669, 200,  // Costume Reginleif Hairband
		19568, 500,  // Costume Ancient Horns
		19678, 500,  // Costume Bell Ribbon
		19750, 500;  // Costume Saint Frill Ribbon

	npcshopdelitem "hunter_shop", 502; 
	for (.@i = 0; .@i < getarraysize(.Shop_Data); .@i += 2) {
		npcshopadditem "hunter_shop", .Shop_Data[.@i], .Shop_Data[.@i+1];
	}
	end;
}
