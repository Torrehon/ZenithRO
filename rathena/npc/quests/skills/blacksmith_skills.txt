geffen,178,72,3	script	Akkie#qsk_bs	726,{
	if (BaseJob != Job_Blacksmith) {
		mes "[Akkie]";
		mes "Ah, today seems like";
		mes "a perfect day for forging.";
		mes "I just feel like getting";
		mes "some high quality materials";
		mes "and hammering away at the anvil!";
		close;
	}

	if (JobLevel < 25) {
		mes "[Akkie]";
		mes "Hello there, "+ strcharinfo(0) +"!";
		mes "I'm Akkie from the Blacksmith Guild.";
		mes "I'm here to teach a powerful new skill to skilled Blacksmiths, but your Job Level is a bit too low right now.";
		next;
		mes "[Akkie]";
		mes "Please train a bit more and come back when you reach ^660000Job Level 25^000000 or higher!";
		close;
	}

	if (getskilllv("BS_DECIMATION") > 0) {
		mes "[Akkie]";
		mes "Hi again, "+ strcharinfo(0) +"!";
		mes "How has your ^0000FFCart Decimation^000000 skill been working for you?";
		mes "Remember to channel your devastating strength into your cart to crush your foes in battle!";
		mes "Keep up the great work and stay safe on your travels!";
		close;
	}

	if (getskilllv("BS_JUGGERNAUTSOUL") > 0 || getskilllv(1047) > 0) {
		mes "[Akkie]";
		mes "...!";
		mes "^333333*Akkie gazes at you in complete awe, backing up slightly from the overwhelming pressure in your aura.*^000000";
		next;
		mes "[Akkie]";
		mes "W-whoa... What incredible, unstoppable strength!";
		mes "The raw physical force in your stance and the devastating power behind your arms is staggering!";
		next;
		mes "[Akkie]";
		mes "I was going to ask you to gather materials, but someone with your brute force has already mastered cart combat beyond anything I could teach.";
		mes "Here, take the knowledge of ^0000FFCart Decimation^000000. It belongs in the hands of a true juggernaut like you!";
		next;
		skill "BS_DECIMATION",1,SKILL_PERM_GRANT;
		set BLACK_SK,2;
		specialeffect EF_LEVELUP;
		mes "[Akkie]";
		mes "Unleash that brutal force upon your enemies! Good luck out there!";
		close;
	}

	if (BLACK_SK == 0) {
		mes "[Akkie]";
		mes "Hello there~";
		mes "I'm Akkie, a representative of the Blacksmith Guild!";
		mes "I'm here to teach Blacksmiths a devastating skill called ^0000FFCart Decimation^000000.";
		next;
		mes "[Akkie]";
		mes "This skill allows you to use your devastating physical strength to deliver a brutal, crushing blow with your pushcart!";
		mes "Would you like me to teach you Cart Decimation?";
		next;
		if (select("Yes, please!:No, thanks.") == 1) {
			mes "[Akkie]";
			mes "Great! You won't regret learning this!";
			mes "To forge the heavy iron reinforcement required for your cart, I will need a few materials from you.";
			next;
			mes "[Akkie]";
			mes "Please bring me:";
			mes "- ^0000FF1 Steel^000000";
			mes "- ^0000FF5 Coal^000000";
			mes "- ^0000FF1 Iron Hammer^000000";
			mes "- ^0000FF1 Detrimindexta^000000";
			mes "- ^0000FF500 Zeny^000000";
			next;
			mes "[Akkie]";
			mes "Once you have all those items, bring them back to me and I'll teach you the skill right away!";
			set BLACK_SK,1;
			close;
		}
		mes "[Akkie]";
		mes "Oh, I see... Well, if you ever change your mind, I'll be right here waiting!";
		close;
	}

	if (BLACK_SK == 1) {
		if (countitem(999) < 1 || countitem(1003) < 5 || countitem(613) < 1 || countitem(971) < 1 || Zeny < 500) {
			mes "[Akkie]";
			mes "Welcome back! Let me check if you brought all the required materials...";
			next;
			mes "[Akkie]";
			mes "Hmm, it looks like you're still missing a few items.";
			mes "Remember, I need:";
			mes "- ^0000FF1 Steel^000000";
			mes "- ^0000FF5 Coal^000000";
			mes "- ^0000FF1 Iron Hammer^000000";
			mes "- ^0000FF1 Detrimindexta^000000";
			mes "- ^0000FF500 Zeny^000000";
			close;
		}

		mes "[Akkie]";
		mes "Awesome! You brought everything!";
		mes "Let me quickly reinforce your equipment using these materials...";
		next;
		mes "^333333*Akkie hammers away passionately at the anvil, crafting heavy iron cart reinforcements.*^000000";
		next;
		delitem 999,1;  // Steel
		delitem 1003,5; // Coal
		delitem 613,1;  // Iron Hammer
		delitem 971,1;  // Detrimindexta
		Zeny -= 500;
		set BLACK_SK,2;
		skill "BS_DECIMATION",1,SKILL_PERM_GRANT;
		specialeffect EF_LEVELUP;
		mes "[Akkie]";
		mes "There you go! You have mastered ^0000FFCart Decimation^000000!";
		mes "Put all your devastating strength behind your cart swings and crush your enemies in battle!";
		close;
	}
}
