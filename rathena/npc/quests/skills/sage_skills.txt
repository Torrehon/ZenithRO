//= [Official Conversion]
//= Quests for skills: Create Converter, Elemental Change (Fire/Earth/Wind/Water)
//============================================================ 

yuno_in03,176,24,3	script	Mischna	755,{
	if (BaseJob != Job_Sage) {
		mes "[Mishuna]";
		mes "Good day, I'm Mishuna, one";
		mes "of the instructors here in the";
		mes "Schweicherbil Magic Academy.";
		mes "By any chance, are you a Sage";
		mes "or Scholar? Oh... You're not?";
		next;
		mes "[Mishuna]";
		mes "Oh, that's too bad.";
		mes "My apologies. But if you";
		mes "happen to know any, or meet";
		mes "any in your journeys, please";
		mes "direct them to me if they haven't heard of the lessons I provide.";
		next;
		mes "[Mishuna]";
		mes "I'm sorry to bother";
		mes "you, and I thank you";
		mes "for your time. Good";
		mes "day to you, adventurer.";
		close;
	}

	// SA_ARCSOUL Shortcut for Create Converter
	if (getskilllv("SA_CREATECON") == 0 && (getskilllv("SA_ARCSOUL") > 0 || getskilllv(1057) > 0)) {
		mes "[Mishuna]";
		mes "...!";
		mes "^333333*Mishuna gazes at you in wonder, sensing a profound elemental resonance.*^000000";
		next;
		mes "[Mishuna]";
		mes "Incredible... Such a profound, harmonious grasp of elemental fundamentals!";
		mes "Your elemental synthesis comprehension is already beyond what any academy manual could teach.";
		next;
		mes "[Mishuna]";
		mes "I don't need to ask you for converter crafting materials. Take the secrets of ^0000FFElemental Converter Creation^000000!";
		next;
		skill "SA_CREATECON",1,SKILL_PERM_GRANT;
		specialeffect2 EF_RUWACH;
		specialeffect EF_LEVELUP;
		mes "[Mishuna]";
		mes "Use this knowledge to craft powerful converters in your research!";
		next;
	}

	mes "[Mishuna]";
	mes "Good day, I'm Mishuna, one";
	mes "of the instructors here in the";
	mes "Schweicherbil Magic Academy.";
	mes "How may I assist your elemental studies today?";
	next;

	while(1) {
		set .@menu_str$, "";
		if (getskilllv("SA_CREATECON") == 0)
			set .@menu_str$, "Learn Elemental Converter Creation:";
		set .@menu_str$, .@menu_str$ + "Learn Elemental Change Skills:Leave";

		set .@choice, select(.@menu_str$);

		if (getskilllv("SA_CREATECON") > 0)
			set .@choice, .@choice + 1;

		if (.@choice == 1) {
			// Learn Elemental Converter Creation (Normal Quest)
			if (countitem(904) < 10 || countitem(947) < 7 || countitem(1013) < 12 || countitem(946) < 10 || countitem(7433) < 4) {
				mes "[Mishuna]";
				mes "To learn ^FF0000Elemental Converter Creation^000000, please bring:";
				next;
				mes "[Mishuna]";
				mes "^ff00007 Horns^000000,";
				mes "^ff000012 Rainbow Shells^000000,";
				mes "^ff000010 Snail's Shells^000000,";
				mes "^ff00004 Blank Scrolls^000000 and";
				mes "^ff000010 Scorpion Tails^000000.";
				close;
			}
			mes "[Mishuna]";
			mes "Great, you brought all the required materials!";
			mes "Please take this copy of Barmundt's scroll and observe closely while I demonstrate...";
			next;
			specialeffect2 EF_ABSORBSPIRITS;
			specialeffect EF_LEVELUP;
			delitem 904,10; // Scorpion_Tail
			delitem 947,7; // Horn
			delitem 1013,12; // Rainbow_Shell
			delitem 946,10; // Snail's_Shell
			delitem 7433,4; // Blank_Scroll
			skill "SA_CREATECON",1,SKILL_PERM_GRANT;
			mes "[Mishuna]";
			mes "Congratulations! You have successfully mastered ^0000FFElemental Converter Creation^000000!";
			close;
		}
		else if (.@choice == 2) {
			// Learn Elemental Change Skills (Allows learning ALL 4!)
			if (getskilllv("SA_CREATECON") == 0) {
				mes "[Mishuna]";
				mes "You must first learn the ^FF0000Elemental Converter Creation^000000 skill before learning Elemental Change skills.";
				close;
			}

			mes "[Mishuna]";
			mes "Sir Barmundt's scrolls contain secrets to change a target's attribute into one of the 4 elements.";
			mes "You may learn ^0000FFall 4 Elemental Change skills^000000 if you meet the requirements!";
			next;

			setarray .@ReqSkill[0], 280, 283, 282, 281;
			setarray .@ReqSkill$[0], "Endow Blaze", "Endow Quake", "Endow Tornado", "Endow Tsunami";
			setarray .@Skill[0], 1018, 1017, 1019, 1008;
			setarray .@Skill$[0], "Fire", "Earth", "Wind", "Water";
			setarray .@ReqItem[0], 990, 993, 992, 991;
			setarray .@ReqItem$[0], "Red Bloods", "Green Lives", "Wind of Verdure", "Crystal Blues";

			while(1) {
				set .@has_fire$, getskilllv("SA_ELEMENTFIRE") > 0 ? " [Learned]" : "";
				set .@has_earth$, getskilllv("SA_ELEMENTGROUND") > 0 ? " [Learned]" : "";
				set .@has_wind$, getskilllv("SA_ELEMENTWIND") > 0 ? " [Learned]" : "";
				set .@has_water$, getskilllv("SA_ELEMENTWATER") > 0 ? " [Learned]" : "";

				set .@i, select("Fire Elemental Change" + .@has_fire$ + ":Earth Elemental Change" + .@has_earth$ + ":Wind Elemental Change" + .@has_wind$ + ":Water Elemental Change" + .@has_water$ + ":Cancel") - 1;

				if (.@i == 4) close;

				if (getskilllv(.@Skill[.@i]) > 0) {
					mes "[Mishuna]";
					mes "You have already mastered ^0000FF" + .@Skill$[.@i] + " Elemental Change^000000!";
					next;
					continue;
				}

				if (getskilllv(.@ReqSkill[.@i]) == 0) {
					mes "[Mishuna]";
					mes "I'm sorry, but you have not learned ^FF0000" + .@ReqSkill$[.@i] + "^000000 yet.";
					mes "You must learn " + .@ReqSkill$[.@i] + " before mastering " + .@Skill$[.@i] + " Elemental Change.";
					next;
					continue;
				}

				if (countitem(.@ReqItem[.@i]) < 40) {
					mes "[Mishuna]";
					mes "To learn ^FF0000" + .@Skill$[.@i] + " Elemental Change^000000, please bring me:";
					next;
					mes "[Mishuna]";
					mes "^FF000040 " + .@ReqItem$[.@i] + "^000000.";
					next;
					continue;
				}

				mes "[Mishuna]";
				mes "Very well! You have brought all the required items for " + .@Skill$[.@i] + " Elemental Change.";
				mes "Please remain still while I chant this incantation...";
				next;
				specialeffect2 EF_RUWACH;
				specialeffect2 EF_BRANDISHSPEAR;
				specialeffect EF_LEVELUP;
				delitem .@ReqItem[.@i], 40;
				skill .@Skill[.@i], 1, SKILL_PERM_GRANT;
				mes "[Mishuna]";
				mes "Congratulations! You have learned ^0000FF" + .@Skill$[.@i] + " Elemental Change^000000!";
				next;
			}
		}
		else {
			mes "[Mishuna]";
			mes "Farewell and good luck with your studies!";
			close;
		}
	}
}
