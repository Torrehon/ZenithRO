payon_in02,64,71,4	script	Archer Master#arc_1	59,{

	// --- EL PORTERO: Filtro anti-intrusos ---
	if (Class != Job_Novice && Class != Job_Novice_High) {
		if (BaseClass == Job_Archer) {
			mes "[Archer Master]";
			mes "Welcome back! Your eyes are as sharp as ever.";
			mes "May your arrows always fly true!";
			close;
		}
		mes "[Archer Master]";
		mes "Your stance is too heavy and your steps make too much noise. The path of the Archer is not for you.";
		close;
	}

	// --- Bloque de Rebirth / Transcendent (High Novice) ---
	if (Class == Job_Novice_High) {
		if (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy) {
			mes "[Archer Master]";
			mes "Hey, I know you. I remember that steady breathing.";
			mes "You took this test before, didn't you?";
			next;
			mes "[Archer Master]";
			mes "Ah, you must have been to Valhalla and been reborn. Wow, that's so impressive!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Master]";
				mes "Err... You'd better learn all the Basic Skills first before you can string a bow again.";
				mes "Alright, see you later.";
				close;
			}
			mes "[Archer Master]";
			mes "Well then. I don't need to say anything else. I know you'll make a great Archer... again!";
			next;
			
			// Cambio de Job (Sin skills automáticas por petición)
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_High;
			
			mes "[Archer Master]";
			mes "Welcome back to the Guild. Now, go show the world why our arrows never miss!";
			close;
		}
		else {
			mes "[Archer Master]";
			mes "I see a reborn soul, but the wind does not guide your hands. Your destiny lies elsewhere.";
			close;
		}
	}

	// --- Nueva Quest Instantánea para Novices ---
	mes "[Archer Master]";
	mes "Good day. Do you wish to join the ranks of the Archers?";
	next;

	// Validamos que tenga Job Nivel 10 y Basic Skills
	if (!callfunc("F_CanChangeJob")) {
		mes "[Archer Master]";
		mes "An Archer needs extremely high concentration and reflexes.";
		next;
		mes "[Archer Master]";
		mes "Your job level must be at least 10, and you must master all of your Basic Skills. Return when you have trained your patience.";
		close;
	}

	mes "[Archer Master]";
	mes "Usually, I would send you out to gather different types of Trunks to prove your patience and eye for quality materials...";
	next;
	mes "[Archer Master]";
	mes "...but the instructors from the Novice Grounds have already sent me a flawless report about your marksmanship.";
	mes "Are you ready to officially join the Archer Guild?";
	next;

	if (select("Yes, I am ready.:Not yet.") == 2) {
		mes "[Archer Master]";
		mes "Hesitation can cost you your life on the battlefield. Come back when your mind is clear.";
		close;
	}

	mes "[Archer Master]";
	mes "Excellent! Let me promote you right away!";
	next;
	
	// Función estándar que hace el cambio de clase y da el equipamiento base
	callfunc "Job_Change", Job_Archer; 
	
	mes "[Archer Master]";
	mes "Congratulations! You are now an Archer!";
	next;
	mes "[Archer Master]";
	mes "Oh, wait a moment! Before you leave for your first hunt, speak to ^0000FFKael^000000 over there.";
	mes "He has been holding onto a very special bow, waiting for a recruit with your exact potential.";
	close;
}
