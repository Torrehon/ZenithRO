geffen_in,164,124,4	script	Mage Master#mag_1	123,{

	// --- EL PORTERO: Filtro anti-intrusos ---
	if (Class != Job_Novice && Class != Job_Novice_High) {
		if (BaseClass == Job_Mage) {
			mes "[Mage Master]";
			mes "Welcome back! One of these days you'll truly realize the power inside of you... when you can spark fire with just your mind!";
			close;
		}
		mes "[Mage Master]";
		mes "What, are you interested in the Mage Guild? I didn't want to tell you this, but your destiny lies in a different path.";
		close;
	}

	// --- Bloque de Rebirth / Transcendent (High Novice) ---
	if (Class == Job_Novice_High) {
		if (ADVJOB == Job_High_Wizard || ADVJOB == Job_Professor) {
			mes "[Mage Master]";
			mes "Whoa, long time no see! But weren't you supposed to be... well, not here?";
			next;
			mes "[Mage Master]";
			mes "Ah, you must have been reborn. Well, I'm glad to have your mystical talent back in the Academy.";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Mage Master]";
				mes "I'm sorry, but I don't think you're ready to channel magic just yet. Go finish mastering your Basic Skills first.";
				close;
			}
			mes "[Mage Master]";
			mes "Since you've walked this path before, I won't question your worth. You want your magic back immediately, don't you?";
			next;
			
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Mage_High;
			
			mes "[Mage Master]";
			mes "Wow, you look even more powerful than before. Go and show the world what a True Mage can do!";
			close;
		}
		else {
			mes "[Mage Master]";
			mes "I see a reborn soul, but the elements do not answer your call. You belong to a different guild.";
			close;
		}
	}

	// --- Nueva Quest Instantánea para Novices ---
	mes "[Mage Master]";
	mes "Yo. What's up? Are you here to tap into the infinite power of the elements?";
	next;

	if (!callfunc("F_CanChangeJob")) {
		mes "[Mage Master]";
		mes "Oh, what a bummer. You haven't met the requirements yet.";
		next;
		mes "[Mage Master]";
		mes "Go back and master your Basic Skills first. Magic requires a solid foundation, you know?";
		close;
	}

	mes "[Mage Master]";
	mes "The instructors at the Novice Grounds sent word of your potential. Usually, I'd make you mix some crazy solutions and gather strange ingredients...";
	next;
	mes "[Mage Master]";
	mes "...but honestly? I can smell the mana radiating from you. You're ready.";
	mes "Welcome to the Geffen Mage Academy! Do you wish to sign the application and become a Mage right now?";
	next;

	if (select("Sign me up!:Not yet.") == 2) {
		mes "[Mage Master]";
		mes "Be a bit more decisive! Magic doesn't wait for the hesitant. Come back when you're ready.";
		close;
	}

	mes "[Mage Master]";
	mes "Excellent! *Ahem*... Congratulations! You are now a Mage!";
	next;
	
	callfunc "Job_Change", Job_Mage; 
	
	mes "[Mage Master]";
	mes "'Welcome to My World~'";
	mes "Heh heh, I just wanted to say that. It's a quote from a movie, you know?";
	next;
	mes "[Mage Master]";
	mes "Oh, before you go to test your new spells! Talk to ^0000FFYuria^000000. She's over there by the wall, and she has a very special staff for a promising rookie like you.";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}

geffen_in,158,124,4	script	Yuria	69,{ //DAWN STAFF QUEST

	// --- EL PORTERO: Filtro anti-intrusos ---
	if (BaseClass != Job_Mage) {
		if (Class == Job_Novice || Class == Job_Novice_High) {
			mes "[Yuria]";
			mes "I can feel a spark in you, but it is still untamed. Go, speak with the Master. He will help you find your focus.";
			close;
		}
		mes "[Yuria]";
		mes "The elements do not whisper to you as they do to us. This tool would be but a simple branch in your hands. Move along.";
		close;
	}

	// === A PARTIR DE AQUÍ, ES 100% RAMA MAGE ===

	if (dawn_staff_q == 2) {
		if (!callfunc("F_HasAnyStarWeapon")) {
			mes "[Yuria]";
			mes "I sense that you do not carry a Star Fragment weapon in your equipment or inventory.";
			mes "If you have lost or broken your Dawn Staff, I can perform a Celestial Re-forge for you.";
			next;
			if (select("Re-forge Dawn Staff (10x Star Dust):Cancel") == 1) {
				callfunc "F_ReForgeDawnWeapon", 40003; // Dawn Wand
			}
			close;
		}
		mes "[Yuria]";
		mes "The mana flows beautifully through that staff. I knew you were the right choice.";
		mes "May your spells always find their mark and your mind remain as clear as crystal.";
		close;
	}

	if (dawn_staff_q == 1) {
		mes "[Yuria]";
		if (JobLevel >= 10) {
			mes "Ah... I can hear it now. The resonance between your soul and the ambient mana is much stronger.";
			mes "You have reached Job Level 10. Your training is bearing fruit.";
			next;
			mes "[Yuria]";
			mes "As promised, here is the ^0000FFDawn Staff^000000. To most, it's just an old, unresponsive piece of wood. But for a student with your potential, it's a key to the arcane.";
			next;
			mes "[Yuria]";
			mes "Take this jewelry box too. It contains a fragment of solidified energy... what commoners call a 'gemstone'.";
			next;
			mes "[Yuria]";
			mes "If your travels take you to the capital, ^0000FFProntera^000000, find the ^0000FFJewel Enchanter^000000 in the refiner's building.";
			mes "He can bind that stone's essence to your staff, awakening its dormant powers. Don't ignore such a gift!";
			next;
			
			getitem 40003, 1; 
			getitem 12106, 1;
			getexp 1000, 1000;
			set dawn_staff_q, 2;
			
			mes "[Yuria]";
			mes "Walk in wisdom, young Mage. The elements are now your allies.";
			close;
		} else {
			mes "The winds tell me you aren't ready. Your control over your internal mana is still too weak.";
			mes "Return when you reach ^FF0000Job Level 10^000000. Only then will you be able to hear the staff's song.";
			close;
		}
	}

	if (dawn_staff_q == 0) {
		mes "[Yuria]";
		mes "A new face in the Academy. The Master says you have the potential to rewrite the laws of reality itself.";
		next;
		mes "[Yuria]";
		mes "I have something that might interest you: the ^0000FFDawn Staff^000000.";
		next;
		mes "[Yuria]";
		mes "Many apprentices have cast it aside, claiming it has no focus and that its wood is 'dead'. They prefer shiny, expensive rods that do the work for them.";
		next;
		mes "[Yuria]";
		mes "But I see a different truth. This staff doesn't give power; it demands it. It requires a Mage who can breathe life back into its grain with their own spirit.";
		next;
		mes "[Yuria]";
		mes "I want to see if you are that person. Hone your magic, reach ^FF0000Job Level 10^000000, and return to me.";
		mes "If you do, the staff shall be yours to awaken. Do we have an accord?";
		next;
		if (select("I will awaken its power.:I'm not ready for this.") == 2) {
			mes "[Yuria]";
			mes "Honesty is a virtue, but hesitation is a wall. Return when you are ready to break it.";
			close;
		}
		mes "[Yuria]";
		mes "Good. I shall be listening for the change in your aura. Do not make me wait too long.";
		set dawn_staff_q, 1;
		close;
	}

OnInit:
	questinfo QTYPE_QUEST, QMARK_PURPLE;
	end;
}
