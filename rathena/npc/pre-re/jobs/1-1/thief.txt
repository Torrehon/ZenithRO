moc_prydb1,39,129,2	script	Thief Master	69,{ //JOB QUEST
	// 1. Verificación si ya es Thief
	if (BaseJob == Job_Thief) {
		mes "[Thief Master]";
		mes "The shadows are your home now. If you ever need guidance, you know where to find me.";
		close;
	}
	
	// 2. Verificación si es de otra clase (no Novice ni Thief)
	else if (BaseJob != Job_Novice && BaseJob != Job_Thief) {
		mes "[Thief Master]";
		mes "You walk a different path. There is nothing for you in the darkness of this guild.";
		close;
	}

	// 3. Flujo principal para Novices
	mes "[Thief Master]";
	mes "I've been watching you...";
	mes "Rumor has it you made it through the Novice Grounds in one piece.";
	next;

	// Validamos que tenga Job Nivel 10 y Basic Skills al máximo
	if (!callfunc("F_CanChangeJob")) {
		mes "[Thief Master]";
		mes "However, it seems you haven't mastered the basics yet.";
		mes "Come back when you have learned all your Basic Skills. The shadows have no mercy for the unprepared.";
		close;
	}

	mes "[Thief Master]";
	mes "Surviving out there requires cunning, light feet, and the will to do whatever it takes.";
	mes "We don't need to send you to steal mushrooms from farms like in the old days. Just making it this far is proof enough of your potential.";
	next;
	mes "[Thief Master]";
	mes "Tell me, are you ready to leave the light behind and embrace the life of a Thief?";
	next;

	if (select("Yes, I am ready.:Not yet.") == 2) {
		mes "[Thief Master]";
		mes "Hah. A smart choice if you're afraid of the dark. Come back when you find your courage.";
		close;
	}

	mes "[Thief Master]";
	mes "Excellent. Welcome to the brotherhood.";
	next;

	// 4. Cambio de Job (Soporte nativo por si un Novice High habla con él)
	if (Class == Job_Novice_High) {
		jobchange Job_Thief_High;
	} else {
		jobchange Job_Thief;
	}

	mes "[Thief Master]";
	mes "Now, disappear into the shadows and make a name for yourself.";
	mes "Oh, and before you go... Speak with Brad over there. He has a special assignment for promising rookies.";
	close;

OnInit:
	questinfo QTYPE_JOB, QMARK_YELLOW;
	end;
}


moc_prydb1,42,133,2	script	Brad	118,{ // DAWN WEAPON QUEST
	// 1. Rechazo a clases que no sean Novice o Thief
	if (BaseJob != Job_Novice && BaseJob != Job_Thief) {
		mes "[Brad]";
		mes "You don't look like you belong in these shadows. Move along.";
		close;
	}

	// 2. Redirección para Novices que aún no han cambiado de Job
	if (BaseJob == Job_Novice) {
		mes "[Brad]";
		mes "If you're looking to join the guild, speak to the Master right next to me.";
		close;
	}

	// === LÓGICA PARA THIEFS ===

	if (dawn_dagger_q == 2) {
		if (!callfunc("F_HasAnyStarWeapon")) {
			mes "[Brad]";
			mes "I sense that you do not carry a Star Fragment weapon in your equipment or inventory.";
			mes "If you have lost or broken your Dawn Dagger, I can perform a Celestial Re-forge for you.";
			next;
			if (select("Re-forge Dawn Dagger (10x Star Dust):Cancel") == 1) {
				callfunc "F_ReForgeDawnWeapon", 40000; // Dawn Stiletto
			}
			close;
		}
		mes "[Brad]";
		mes "How's the Dawn Dagger treating you? I knew you'd be the one to finally make that blade sing.";
		close;
	}

	// Si la quest está en curso
	if (dawn_dagger_q == 1) {
		mes "[Brad]";
		// Verificamos si ya alcanzó el Job Level 10
		if (JobLevel >= 10) {
			mes "Well, well. Job Level 10. You really are a fast learner.";
			next;
			mes "[Brad]";
			mes "As promised, here is the blade. Most see it as a useless piece of scrap, but let's see what you can do with it.";
			next;
			mes "[Brad]";
			mes "I'm also giving you this small jewelry box. It contains a random gemstone of considerable value.";
			next;
			mes "[Brad]";
			mes "If you ever find yourself in the capital, ^0000FFProntera^000000, visit the refiner's building. There is a man there known as the ^0000FFJewel Enchanter^000000.";
			mes "He has the ability to use these jewels to awaken hidden properties in your equipment. It might be exactly what you need to push that dagger to its limit.";
			next;
			
			// Entregamos la Daga Custom, la Caja de Joyas, la EXP y actualizamos la quest
			getitem 40000, 1; 
			getitem 12106, 1;
			getexp 1000, 1000;
			set dawn_dagger_q, 2;
			
			mes "[Brad]";
			mes "Wield the ^0000FFDawn Dagger^000000 well. Show the world what true potential looks like.";
			close;
		} else {
			mes "You're not quite there yet. You need to reach ^FF0000Job Level 10^000000 before I hand over the blade.";
			mes "Keep training. Show me you have the skill to awaken its potential.";
			close;
		}
	}

	// Si la quest aún no ha sido iniciada
	if (dawn_dagger_q == 0) {
		mes "[Brad]";
		mes "Ah, the new recruit. The Master mentioned you.";
		next;
		mes "[Brad]";
		mes "I have a rather... peculiar item in my inventory. It's called the ^0000FFDawn Dagger^000000.";
		next;
		mes "[Brad]";
		mes "To be perfectly honest, to a normal person, it's completely useless. It's oddly weighted, the edge seems dull, and most recruits just throw it back in my face.";
		next;
		mes "[Brad]";
		mes "But watching you move... I see something different. A spark of potential. I suspect that in the right hands, this 'junk' might just be the deadliest weapon in the shadows.";
		next;
		mes "[Brad]";
		mes "I want to test that theory. Get out there, hone your skills, and reach ^FF0000Job Level 10^000000.";
		mes "Do that, and the dagger is yours to master. What do you say?";
		next;
		if (select("I'll show you its true power.:Maybe later.") == 2) {
			mes "[Brad]";
			mes "A shame. Let me know if you decide you're up for the challenge.";
			close;
		}
		mes "[Brad]";
		mes "Good. I'll be watching your progress. Prove my instincts right.";
		set dawn_dagger_q, 1;
		close;
	}

OnInit:
	questinfo QTYPE_QUEST, QMARK_PURPLE;
	end;
}
