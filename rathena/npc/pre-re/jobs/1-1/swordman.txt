izlude_in,74,172,4	script	Swordman Master#swd_1	119,{ // SWORDMAN QUEST
	
	// --- EL PORTERO: Filtro anti-intrusos ---
	if (Class != Job_Novice && Class != Job_Novice_High) {
		if (BaseClass == Job_Swordman) {
			mes "[Swordman Master]";
			mes "Job change? Muhahaha! But you're already walking the path of the Sword! Be proud and be strong!";
			close;
		}
		mes "[Swordman Master]";
		mes "I only have business with Novices. You already chose your path in life. Be gone!";
		close;
	}

	// --- Bloque de Rebirth / Transcendent (High Novice) ---
	if (Class == Job_Novice_High) {
		if (ADVJOB == Job_Lord_Knight || ADVJOB == Job_Paladin) {
			mes "[Swordman Master]";
			mes "It...";
			mes "Can't be...";
			mes "You've been reborn, haven't you?";
			next;
			mes "[Swordman Master]";
			mes "I see you're retreading the path of the Swordman! Once you've gotten used to brandishing a sword, you can never go back!!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Swordman Master]";
				mes "Hmm? Ah, you must first master the Basic Skills before you are ready to become a Swordman.";
				next;
				mes "[Swordman Master]";
				mes "Come back to me when you have finished learning the Basic Novice Skills.";
				close;
			}
			mes "[Swordman Master]";
			mes "Excellent! Let me promote you to a Swordman right away!";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Swordman_High;
			skill "SM_MOVINGRECOVERY",1,SKILL_PERM;
			skill "SM_FATALBLOW",1,SKILL_PERM;
			skill "SM_AUTOBERSERK",1,SKILL_PERM;
			mes "[Swordman Master]";
			mes "Hmm... You look like a well-experienced Swordman. Still, I'm sure that you must train to improve your skills and gain strength!";
			close;
		}
		else {
			mes "[Swordman Master]";
			mes "Hm...? You're a reborn warrior, aren't you?";
			next;
			mes "[Swordman Master]";
			mes "Hmmm... It seems that being a Swordman is not part of your destiny. I'm sorry.";
			close;
		}
	}

	// --- Nueva Quest Instantánea para Novices ---
	mes "[Swordman Master]";
	mes "Welcome to the Swordman Association!";
	mes "I have received word of your arrival. The instructors at the Novice Grounds speak highly of your strength!";
	next;

	if (!callfunc("F_CanChangeJob")) {
		mes "[Swordman Master]";
		mes "However, I see you still haven't fully mastered your Basic Skills. A true warrior must have a solid foundation!";
		close;
	}

	mes "[Swordman Master]";
	mes "Are you ready to cast aside your past and walk the proud path of the Swordman?";
	next;

	if (select("Yes, I am ready.:I need more time.") == 2) {
		mes "[Swordman Master]";
		mes "Return when your heart is resolute!";
		close;
	}

	mes "[Swordman Master]";
	mes "Excellent! Your journey begins right now!";
	next;
	
	callfunc "Job_Change", Job_Swordman;
	
	mes "[Swordman Master]";
	mes "Congratulations! You are now a proud Swordman.";
	next;
	mes "[Swordman Master]";
	mes "Before you depart, speak with ^0000FFVeteran Leon^000000 over there. He has a challenge for you.";
	close;
}
